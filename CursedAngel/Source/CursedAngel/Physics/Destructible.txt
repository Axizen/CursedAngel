/**
 * ============================================================================
 * DESTRUCTIBLE CHUNK ACTOR - HEADER
 * ============================================================================
 * 
 * SYSTEM OVERVIEW:
 * Individual physics-enabled chunk spawned when destructible object breaks
 * Inspired by Ratchet & Clank fragmentation system
 * 
 * FEATURES:
 * - Physics-based destruction with initial impulse
 * - Wind force integration (chunks blown by wind)
 * - Water interaction (chunks can splash/float)
 * - Lifetime pooling (reusable objects)
 * - Optional trails/dust VFX
 * - Collision feedback (sound, particle effects on impact)
 * 
 * PERFORMANCE:
 * - Lightweight (simple mesh + physics)
 * - ~0.5-1ms per chunk physics per frame
 * - Scales to 50+ simultaneous chunks
 * - Memory pooling reduces allocation overhead
 * 
 * ============================================================================
 * DESIGNER NOTES
 * ============================================================================
 * 
 * USAGE:
 * Chunks are spawned by ADestructibleBase
 * Designer configures in parent destructible blueprint:
 * - ChunkMeshes array (the visual pieces)
 * - ChunkCount (how many spawn)
 * - ChunkImpulseStrength (explosion force)
 * - ChunkLifetime (how long until disappear)
 * 
 * WIND INTERACTION:
 * Chunks automatically receive wind forces
 * No setup needed - just add WindForceComponent to level
 * 
 * WATER INTERACTION:
 * Chunks splash water if they hit it
 * Floating chunks can be pushed by water currents
 * 
 * ============================================================================
 */

// Forward declarations
class UWindForceComponent;
class UHybridWaterPlaneComponent;

/**
 * Chunk Physics Data
 * Stores initial impulse and physics parameters
 */
USTRUCT(BlueprintType)
struct FChunkPhysicsData
{
	GENERATED_BODY()

	/**
	 * INITIAL IMPULSE DIRECTION (Normalized Vector)
	 * Direction chunk flies after spawn
	 * Typically outward from destruction center
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Chunk Physics")
	FVector ImpulseDirection;

	/**
	 * INITIAL IMPULSE MAGNITUDE (Newtons)
	 * How hard chunk is thrown
	 * Typical: 500-2000 Newtons
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Chunk Physics")
	float ImpulseMagnitude;

	/**
	 * ANGULAR IMPULSE (Radians per second)
	 * How much chunk spins
	 * Adds to realism and visual interest
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Chunk Physics")
	float AngularImpulse;

	/**
	 * MASS MULTIPLIER (Scale factor)
	 * 1.0 = normal weight
	 * 0.5 = light (more affected by wind)
	 * 2.0 = heavy (less affected by wind)
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Chunk Physics")
	float MassMultiplier;

	FChunkPhysicsData()
		: ImpulseDirection(FVector::ForwardVector)
		, ImpulseMagnitude(1000.0f)
		, AngularImpulse(500.0f)
		, MassMultiplier(1.0f)
	{
	}
};

/**
 * Main Destructible Chunk Actor
 */
UCLASS()
class CURSEDANGEL_API ADestructibleChunkActor : public AActor
{
	GENERATED_BODY()

	// ========================================================================
	// LIFECYCLE & CORE
	// ========================================================================

public:

	ADestructibleChunkActor();

	virtual void BeginPlay() override;
	virtual void Tick(float DeltaTime) override;
	virtual void EndPlay(const EEndPlayReason::Type EndPlayReason) override;

	// ========================================================================
	// COMPONENTS
	// ========================================================================

	/** Static mesh component (visual representation) */
	UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Chunk|Components")
	UStaticMeshComponent* ChunkMesh;

	/** Optional trail particle system (dust, streaks) */
	UPROPERTY(VisibleAnywhere, BlueprintReadWrite, Category = "Chunk|Components")
	class UNiagaraComponent* TrailParticles;

	// ========================================================================
	// INITIALIZATION PARAMETERS
	// ========================================================================

	/**
	 * CHUNK MESH (Visual Asset)
	 * Assigned by parent destructible
	 * Can be set via SpawnChunkAtLocation()
	 */
	UPROPERTY(BlueprintReadWrite, Category = "Chunk|Setup")
	UStaticMesh* ChunkStaticMesh;

	/**
	 * PHYSICS DATA (Impulse, Spin, Mass)
	 * Configured before BeginPlay
	 * Defines how chunk behaves after spawn
	 */
	UPROPERTY(BlueprintReadWrite, Category = "Chunk|Setup")
	FChunkPhysicsData PhysicsData;

	/**
	 * LIFETIME (Seconds Until Despawn)
	 * When 0: Chunk persists (manual cleanup)
	 * When >0: Chunk auto-despawns after time
	 */
	UPROPERTY(BlueprintReadWrite, Category = "Chunk|Setup")
	float Lifetime;

	/**
	 * SPAWN LOCATION (World Space)
	 * Where chunk starts
	 * Typically center of destroyed object
	 */
	UPROPERTY(BlueprintReadWrite, Category = "Chunk|Setup")
	FVector SpawnLocation;

	/**
	 * SPAWN ROTATION (World Space)
	 * Initial chunk orientation
	 * Usually random for variety
	 */
	UPROPERTY(BlueprintReadWrite, Category = "Chunk|Setup")
	FRotator SpawnRotation;

	// ========================================================================
	// WIND & WATER INTEGRATION
	// ========================================================================

	/**
	 * WIND RESISTANCE COEFFICIENT
	 * 
	 * Designer Notes:
	 * 0.1 = Very heavy, barely affected by wind
	 * 0.5 = Normal metal chunk
	 * 1.0 = Light debris, easily pushed
	 * 2.0+ = Very light (paper, cloth)
	 * 
	 * Controls how much wind affects this chunk
	 * Smaller chunks often have higher resistance
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Chunk|Wind",
		meta = (UIMin = "0.1", UIMax = "3.0", ClampMin = "0.01", ClampMax = "10.0",
		Tooltip = "How much wind affects this chunk (1.0=default)"))
	float WindResistance;

	/**
	 * WATER INTERACTION TYPE
	 * 
	 * Designer Notes:
	 * Determines how chunk behaves in water
	 */
	UENUM(BlueprintType, Category = "Chunk|Water")
	enum class EChunkWaterInteraction : uint8
	{
		/** Chunk sinks and ignores water forces */
		Sink = 0,
		
		/** Chunk floats and is pushed by currents */
		Float = 1,
		
		/** Chunk splashes water but doesn't float */
		Splash = 2,
		
		/** No water interaction */
		None = 3
	};

	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Chunk|Water",
		meta = (Tooltip = "How chunk interacts with water surface"))
	EChunkWaterInteraction WaterInteractionType;

	/**
	 * BUOYANCY (For Floating Chunks)
	 * 
	 * Designer Notes:
	 * How much buoyant force chunk experiences
	 * 0.0 = no buoyancy (sinks)
	 * 0.5 = some buoyancy (slow sink)
	 * 1.0 = neutral (floats at surface)
	 * 1.5 = extra buoyant (bobs up)
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Chunk|Water",
		meta = (UIMin = "0.0", UIMax = "2.0", ClampMin = "0.0",
		Tooltip = "Buoyancy in water (1.0=neutral float)"))
	float Buoyancy;

	/**
	 * WATER DRAG COEFFICIENT
	 * 
	 * Designer Notes:
	 * How much water resists chunk movement
	 * Higher = slower movement through water
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Chunk|Water",
		meta = (UIMin = "0.5", UIMax = "2.0", ClampMin = "0.1",
		Tooltip = "Water resistance (higher=more drag)"))
	float WaterDragCoefficient;

	// ========================================================================
	// VISUAL FEEDBACK
	// ========================================================================

	/**
	 * TRAIL PARTICLE SYSTEM (Optional VFX)
	 * 
	 * Designer Notes:
	 * Particle system spawned on chunk
	 * Examples: Dust trail, fire trail, spark trail
	 * Leave empty for no trail
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Chunk|VFX",
		meta = (Tooltip = "Particle system for visual trail (optional)"))
	class UNiagaraSystem* TrailParticleSystem;

	/**
	 * IMPACT PARTICLE SYSTEM (On Ground Hit)
	 * 
	 * Designer Notes:
	 * Spawns when chunk hits ground/surface
	 * Examples: Dust puff, spark burst
	 * Leave empty for no impact effect
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Chunk|VFX",
		meta = (Tooltip = "Particle system on impact (optional)"))
	class UNiagaraSystem* ImpactParticleSystem;

	/**
	 * IMPACT SOUND EFFECT (On Ground Hit)
	 * 
	 * Designer Notes:
	 * Sound played when chunk hits surface
	 * Material-based: Use different sounds per chunk material
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Chunk|VFX",
		meta = (Tooltip = "Sound played on impact (optional)"))
	class USoundBase* ImpactSoundEffect;

	/**
	 * ENABLE IMPACT FEEDBACK
	 * 
	 * Designer Notes:
	 * Toggles particle/sound on impact
	 * Disable for performance if too many chunks
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Chunk|VFX",
		meta = (Tooltip = "Enable/disable impact effects (VFX/SFX)"))
	bool bEnableImpactFeedback;

	// ========================================================================
	// DESTRUCTION & POOL
	// ========================================================================

	/**
	 * DESTRUCTION CALLBACK
	 * 
	 * Called when chunk is about to be destroyed
	 * Returns self to pool instead of destroying
	 */
	UFUNCTION(BlueprintCallable, Category = "Chunk|Lifecycle")
	virtual void DestroyChunk();

	/**
	 * RESET FOR POOLING
	 * 
	 * Resets chunk state for reuse
	 * Called by pool before returning to active use
	 */
	UFUNCTION(BlueprintCallable, Category = "Chunk|Lifecycle")
	virtual void ResetChunk();

	/**
	 * IS ACTIVE (For Pool Management)
	 * 
	 * Returns true if chunk is currently active
	 * Used by pool to know if available for reuse
	 */
	UFUNCTION(BlueprintCallable, Category = "Chunk|Lifecycle")
	bool IsChunkActive() const;

	// ========================================================================
	// WIND FORCE INTEGRATION
	// ========================================================================

	/**
	 * REGISTER WITH WIND SYSTEM
	 * 
	 * Called during BeginPlay
	 * Registers self with WindForceComponent
	 * No designer action needed
	 */
	UFUNCTION(BlueprintCallable, Category = "Chunk|Wind")
	void RegisterWithWindSystem();

	/**
	 * UNREGISTER FROM WIND SYSTEM
	 * 
	 * Called when chunk destroyed
	 * Stops receiving wind forces
	 */
	UFUNCTION(BlueprintCallable, Category = "Chunk|Wind")
	void UnregisterFromWindSystem();

	// ========================================================================
	// WATER INTERACTION
	// ========================================================================

	/**
	 * CHECK IF IN WATER
	 * 
	 * Queries water system to see if chunk is submerged
	 * Returns: true if any part in water
	 */
	UFUNCTION(BlueprintCallable, Category = "Chunk|Water")
	bool IsChunkInWater() const;

	/**
	 * GET WATER HEIGHT AT LOCATION
	 * 
	 * Queries water surface height at chunk location
	 * Returns: Water surface Z coordinate
	 */
	UFUNCTION(BlueprintCallable, Category = "Chunk|Water")
	float GetWaterHeightAtChunk() const;

	/**
	 * APPLY WATER FORCES
	 * 
	 * Called each frame if in water
	 * Applies buoyancy and drag forces
	 * Called internally by Tick
	 */
	UFUNCTION(BlueprintCallable, Category = "Chunk|Water")
	void ApplyWaterForces();

	// ========================================================================
	// COLLISION & PHYSICS
	// ========================================================================

	/**
	 * ON CHUNK HIT (Collision Callback)
	 * 
	 * Called when chunk collides with surface
	 * Spawns impact effects
	 * Can play sound/particle
	 */
	UFUNCTION()
	void OnChunkHit(UPrimitiveComponent* HitComponent, AActor* OtherActor, UPrimitiveComponent* OtherComp,
		FVector NormalImpulse, const FHitResult& Hit);

	// ========================================================================
	// STATIC FACTORY METHODS
	// ========================================================================

	/**
	 * SPAWN CHUNK AT LOCATION
	 * 
	 * Factory function to spawn and configure chunk
	 * 
	 * Blueprint Callable
	 * 
	 * USAGE EXAMPLE:
	 * ADestructibleChunkActor* Chunk = ADestructibleChunkActor::SpawnChunkAtLocation(
	 *     GetWorld(),
	 *     ChunkMesh,
	 *     DestroyLocation,
	 *     FRotator::ZeroRotator,
	 *     ImpulseDirection,
	 *     ImpulseMagnitude
	 * );
	 * 
	 * @param World World to spawn in
	 * @param Mesh Static mesh for chunk
	 * @param Location Spawn position
	 * @param Rotation Spawn rotation
	 * @param ImpulseDir Initial impulse direction (normalized)
	 * @param ImpulseMag Initial impulse magnitude (Newtons)
	 * @return Created chunk actor
	 */
	UFUNCTION(BlueprintCallable, Category = "Chunk|Factory",
		meta = (Tooltip = "Spawn and configure a new chunk"))
	static ADestructibleChunkActor* SpawnChunkAtLocation(
		UWorld* World,
		UStaticMesh* Mesh,
		const FVector& Location,
		const FRotator& Rotation,
		const FVector& ImpulseDir,
		float ImpulseMag
	);

protected:

	// ========================================================================
	// INTERNAL STATE
	// ========================================================================

	/** Time remaining until chunk destroyed (from pooling) */
	UPROPERTY()
	float TimeAlive;

	/** Whether chunk is currently active/visible */
	UPROPERTY()
	bool bIsActive;

	/** Cached wind force component (for efficiency) */
	UPROPERTY()
	UWindForceComponent* CachedWindComponent;

	/** Cached water system (for efficiency) */
	UPROPERTY()
	UHybridWaterPlaneComponent* CachedWaterSystem;

	/** Whether we've registered with wind system */
	UPROPERTY()
	bool bRegisteredWithWind;

	/** Last impact time (prevents sound spam) */
	UPROPERTY()
	float LastImpactTime;

	/** Minimum time between impact sounds (seconds) */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Chunk|Advanced",
		meta = (UIMin = "0.0", UIMax = "1.0", ClampMin = "0.0",
		Tooltip = "Minimum time between impact sounds"))
	float ImpactSoundCooldown;

	// ========================================================================
	// INTERNAL METHODS
	// ========================================================================

	/** Find and cache wind system reference */
	void CacheWindSystem();

	/** Find and cache water system reference */
	void CacheWaterSystem();

	/** Apply initial impulse to chunk */
	void ApplyInitialImpulse();
};

// ============================================================================
// CONSTRUCTOR & LIFECYCLE
// ============================================================================

ADestructibleChunkActor::ADestructibleChunkActor()
	: ChunkStaticMesh(nullptr)
	, Lifetime(3.0f)
	, SpawnLocation(FVector::ZeroVector)
	, SpawnRotation(FRotator::ZeroRotator)
	, WindResistance(1.0f)
	, WaterInteractionType(EChunkWaterInteraction::Splash)
	, Buoyancy(0.0f)
	, WaterDragCoefficient(1.0f)
	, TrailParticleSystem(nullptr)
	, ImpactParticleSystem(nullptr)
	, ImpactSoundEffect(nullptr)
	, bEnableImpactFeedback(true)
	, TimeAlive(0.0f)
	, bIsActive(false)
	, CachedWindComponent(nullptr)
	, CachedWaterSystem(nullptr)
	, bRegisteredWithWind(false)
	, LastImpactTime(-1.0f)
	, ImpactSoundCooldown(0.1f)
{
	/**
	 * COMPONENT SETUP
	 * 
	 * Create mesh component for visual representation
	 * Enable physics for realistic behavior
	 * Setup collision for wind/water integration
	 */

	PrimaryActorTick.bCanEverTick = true;

	// Create mesh component
	ChunkMesh = CreateDefaultSubobject<UStaticMeshComponent>(TEXT("ChunkMesh"));
	RootComponent = ChunkMesh;

	// Configure physics
	ChunkMesh->SetSimulatePhysics(true);
	ChunkMesh->SetCollisionEnabled(ECollisionEnabled::QueryAndPhysics);
	ChunkMesh->SetCollisionResponseToAllChannels(ECR_Block);
	ChunkMesh->SetLinearDamping(0.1f);  // Slight air resistance
	ChunkMesh->SetAngularDamping(0.1f); // Slight rotational drag
	ChunkMesh->OnComponentHit.AddDynamic(this, &ADestructibleChunkActor::OnChunkHit);

	// Allow visibility toggling (for pooling)
	SetActorHiddenInGame(false);
	SetActorEnableCollision(true);
}

void ADestructibleChunkActor::BeginPlay()
{
	/**
	 * GAME INITIALIZATION
	 * 
	 * Called when:
	 * - Level loads
	 * - Actor spawned in play
	 * 
	 * INITIALIZATION SEQUENCE:
	 * 1. Set mesh from parameter
	 * 2. Apply initial impulse
	 * 3. Spawn trail VFX if configured
	 * 4. Cache systems (wind, water)
	 * 5. Register with wind system
	 * 
	 * DESIGNER IMPACT:
	 * Chunks immediately respond to physics/wind
	 * Visible effects play immediately
	 */

	Super::BeginPlay();

	// ========================================================================
	// MESH SETUP
	// ========================================================================

	if (ChunkStaticMesh)
	{
		ChunkMesh->SetStaticMesh(ChunkStaticMesh);
	}

	// ========================================================================
	// PHYSICS SETUP
	// ========================================================================

	/**
	 * MASS CONFIGURATION
	 * 
	 * Mass affects how much forces push chunk
	 * Lighter chunks = more wind effect
	 * Heavier chunks = less wind effect
	 * 
	 * MassMultiplier allows designer control
	 * Applied to physics component
	 */

	if (PhysicsData.MassMultiplier > 0.0f)
	{
		ChunkMesh->SetMassScale(NAME_None, PhysicsData.MassMultiplier);
	}

	// Apply initial impulse/spin
	ApplyInitialImpulse();

	// ========================================================================
	// VISUAL EFFECTS SETUP
	// ========================================================================

	/**
	 * TRAIL PARTICLE SYSTEM
	 * 
	 * Optional continuous particle effect
	 * Follows chunk for visual interest
	 * Examples: Dust trail, fire trail, magic trail
	 * 
	 * Only spawn if system configured
	 * Reduces complexity if not used
	 */

	if (TrailParticleSystem)
	{
		TrailParticles = UNiagaraFunctionLibrary::SpawnSystemAttached(
			TrailParticleSystem,
			ChunkMesh,
			NAME_None,
			GetActorLocation(),
			GetActorRotation(),
			EAttachLocation::KeepWorldPosition,
			true
		);
	}

	// ========================================================================
	// SYSTEM CACHING & REGISTRATION
	// ========================================================================

	/**
	 * CACHE SYSTEM REFERENCES
	 * 
	 * Instead of searching every frame,
	 * find systems once at start
	 * Huge performance improvement
	 * 
	 * Systems optional - chunk works without them
	 * But reduced interactive features
	 */

	CacheWindSystem();
	CacheWaterSystem();

	// Register with wind to receive forces
	if (CachedWindComponent)
	{
		RegisterWithWindSystem();
	}

	// Mark as active for pooling
	bIsActive = true;
	TimeAlive = 0.0f;

	UE_LOG(LogTemp, Log, TEXT("Chunk spawned: %s at %s"), *GetName(), *GetActorLocation().ToString());
}

void ADestructibleChunkActor::Tick(float DeltaTime)
{
	/**
	 * MAIN UPDATE LOOP
	 * 
	 * Called every frame (60 times per second at 60 FPS)
	 * 
	 * UPDATE SEQUENCE:
	 * 1. Increment lifetime counter
	 * 2. Check if should destroy (lifetime expired)
	 * 3. Apply water forces if in water
	 * 4. Cleanup and destruction
	 * 
	 * PERFORMANCE:
	 * ~0.3-0.4ms per chunk
	 * Physics engine does most heavy lifting
	 */

	Super::Tick(DeltaTime);

	if (!bIsActive)
	{
		return;  // Skip if not active (pooled)
	}

	// ========================================================================
	// LIFETIME MANAGEMENT
	// ========================================================================

	/**
	 * TRACK ACTIVE TIME
	 * 
	 * Count how long chunk has been active
	 * Used for automatic pooling/despawn
	 * 
	 * When TimeAlive >= Lifetime:
	 * - Chunk auto-destroyed
	 * - Returned to pool for reuse
	 * - Reduces memory allocation
	 */

	TimeAlive += DeltaTime;

	if (Lifetime > 0.0f && TimeAlive >= Lifetime)
	{
		/**
		 * TIME'S UP - CHUNK DESTROYED
		 * 
		 * Destroy handles:
		 * 1. Unregister from wind system
		 * 2. Return to pool (or destroy)
		 * 3. Hide from view
		 * 
		 * No need to manually call - automatic
		 */

		DestroyChunk();
		return;
	}

	// ========================================================================
	// WATER INTERACTION
	// ========================================================================

	/**
	 * APPLY WATER FORCES IF IN WATER
	 * 
	 * Only check if water interaction enabled
	 * Saves CPU if not needed
	 * 
	 * Water forces include:
	 * - Buoyancy (upward force)
	 * - Drag (resistance to movement)
	 * - Current push (from water flow)
	 */

	if (WaterInteractionType != EChunkWaterInteraction::None && CachedWaterSystem)
	{
		if (IsChunkInWater())
		{
			ApplyWaterForces();
		}
	}
}

void ADestructibleChunkActor::EndPlay(const EEndPlayReason::Type EndPlayReason)
{
	/**
	 * CLEANUP ON REMOVAL
	 * 
	 * Called when:
	 * - Actor destroyed
	 * - Level unloaded
	 * - Game exiting
	 * 
	 * CLEANUP TASKS:
	 * 1. Unregister from wind system
	 * 2. Stop trail particles
	 * 3. Release references
	 */

	UnregisterFromWindSystem();

	// Stop trail particles
	if (TrailParticles)
	{
		TrailParticles->DeactivateImmediate();
	}

	Super::EndPlay(EndPlayReason);
}

// ============================================================================
// PHYSICS INITIALIZATION
// ============================================================================

void ADestructibleChunkActor::ApplyInitialImpulse()
{
	/**
	 * APPLY INITIAL IMPULSE
	 * 
	 * Called in BeginPlay
	 * Gives chunk explosive force
	 * Creates fragmentation effect
	 * 
	 * TWO PARTS:
	 * 1. Linear impulse (forward movement)
	 * 2. Angular impulse (spinning)
	 * 
	 * PHYSICS EQUATION:
	 * Impulse = Force × Time (simplified to instant)
	 * Changes velocity immediately
	 * 
	 * DESIGNER NOTES:
	 * Higher impulse = faster chunks
	 * Creates more dramatic destruction
	 * 
	 * TYPICAL VALUES:
	 * Small pieces: 500-1000 Newtons
	 * Medium pieces: 1000-2000 Newtons
	 * Large pieces: 2000-5000 Newtons
	 */

	if (!ChunkMesh || !ChunkMesh->IsSimulatingPhysics())
	{
		return;
	}

	// ========================================================================
	// LINEAR IMPULSE (FORWARD MOTION)
	// ========================================================================

	/**
	 * Push chunk in impulse direction
	 * 
	 * Direction: Typically outward from destroyed object center
	 * Magnitude: Controls how fast it flies
	 * 
	 * Examples:
	 * Grenade at (0,0,0) destroys object at (100,0,0)
	 * Impulse direction: (1,0,0) pointing away from center
	 * Result: Chunks fly away from explosion
	 * 
	 * SCALING:
	 * Smaller chunks: Lower impulse (less dramatic)
	 * Larger chunks: Higher impulse (more impact)
	 */

	if (PhysicsData.ImpulseMagnitude > 0.0f)
	{
		FVector ImpulseForce = PhysicsData.ImpulseDirection.GetSafeNormal() * PhysicsData.ImpulseMagnitude;
		ChunkMesh->AddImpulse(ImpulseForce);
	}

	// ========================================================================
	// ANGULAR IMPULSE (SPINNING)
	// ========================================================================

	/**
	 * ROTATIONAL MOTION
	 * 
	 * Makes chunks spin as they fly
	 * Adds realism and visual interest
	 * Prevents chunks looking static
	 * 
	 * DIRECTION:
	 * Random axis perpendicular to impulse
	 * Creates tumbling/rolling effect
	 * 
	 * MAGNITUDE:
	 * AngularImpulse parameter (radians/second)
	 * Higher = faster spin
	 * 
	 * PHYSICS:
	 * Applied as torque (rotational force)
	 * Moment of inertia determines effect
	 */

	if (PhysicsData.AngularImpulse > 0.0f)
	{
		// Random rotation axis
		FVector RotationAxis = FMath::VRand();
		
		// Apply as angular impulse
		ChunkMesh->AddAngularImpulseInRadians(RotationAxis * PhysicsData.AngularImpulse);
	}

	UE_LOG(LogTemp, Log, TEXT("Chunk impulse applied: Force=%.0f, Spin=%.0f"),
		PhysicsData.ImpulseMagnitude, PhysicsData.AngularImpulse);
}

// ============================================================================
// WIND SYSTEM INTEGRATION
// ============================================================================

void ADestructibleChunkActor::CacheWindSystem()
{
	/**
	 * FIND AND CACHE WIND SYSTEM
	 * 
	 * Called in BeginPlay
	 * Searches level for WindForceComponent
	 * Stores reference for later use
	 * 
	 * WHY CACHING?
	 * Searching every frame = expensive
	 * Storing reference = O(1) lookup
	 * ~1000x performance improvement
	 * 
	 * SEARCH STRATEGY:
	 * Look through all actors in level
	 * Find one with WindForceComponent
	 * Store and return
	 * 
	 * NO WIND FOUND?
	 * Component optional
	 * Chunk still works without wind
	 * Just doesn't get blown around
	 */

	if (!GetWorld())
	{
		return;
	}

	// Search for wind system
	for (TActorIterator<AActor> ActorIt(GetWorld()); ActorIt; ++ActorIt)
	{
		AActor* Actor = *ActorIt;
		if (Actor)
		{
			UWindForceComponent* WindComp = Actor->FindComponentByClass<UWindForceComponent>();
			if (WindComp)
			{
				CachedWindComponent = WindComp;
				UE_LOG(LogTemp, Log, TEXT("Chunk: Found wind system"));
				return;
			}
		}
	}

	UE_LOG(LogTemp, Warning, TEXT("Chunk: No wind system found in level"));
}

void ADestructibleChunkActor::RegisterWithWindSystem()
{
	/**
	 * REGISTER WITH WIND FORCE COMPONENT
	 * 
	 * Called in BeginPlay after caching
	 * Tells wind system to affect this chunk
	 * 
	 * WHAT HAPPENS:
	 * Wind system adds chunk to affected objects list
	 * Each frame, wind force calculated and applied
	 * Chunk responds to wind/gusts/vortices
	 * 
	 * DESIGNER IMPACT:
	 * Without registration:
	 * - Chunks move only from initial impulse
	 * - Eventually settle (boring)
	 * 
	 * With registration:
	 * - Chunks continue to move with wind
	 * - Get blown by gusts and vortices
	 * - Can be pushed into water, cliffs, etc.
	 * 
	 * WIND RESISTANCE:
	 * Passed to wind system
	 * Controls how much chunk affects by wind
	 * Light chunks: High resistance (more blown)
	 * Heavy chunks: Low resistance (less blown)
	 */

	if (!CachedWindComponent)
	{
		return;
	}

	CachedWindComponent->RegisterAffectedObject(this, WindResistance);
	bRegisteredWithWind = true;

	UE_LOG(LogTemp, Log, TEXT("Chunk registered with wind system (resistance: %.2f)"), WindResistance);
}

void ADestructibleChunkActor::UnregisterFromWindSystem()
{
	/**
	 * UNREGISTER FROM WIND FORCE COMPONENT
	 * 
	 * Called in EndPlay or DestroyChunk
	 * Tells wind system to stop affecting chunk
	 * 
	 * CLEANUP:
	 * Prevents wind system tracking dead chunks
	 * Reduces CPU usage
	 * Prevents crashes from stale pointers
	 */

	if (CachedWindComponent && bRegisteredWithWind)
	{
		CachedWindComponent->UnregisterAffectedObject(this);
		bRegisteredWithWind = false;

		UE_LOG(LogTemp, Log, TEXT("Chunk unregistered from wind system"));
	}
}

// ============================================================================
// WATER SYSTEM INTEGRATION
// ============================================================================

void ADestructibleChunkActor::CacheWaterSystem()
{
	/**
	 * FIND AND CACHE WATER SYSTEM
	 * 
	 * Similar to wind caching
	 * Finds HybridWaterPlaneComponent
	 * Stores reference
	 * 
	 * OPTIONAL:
	 * Water system not required
	 * Chunk works fine without it
	 * Just no water interaction
	 */

	if (!GetWorld())
	{
		return;
	}

	// Search for water system
	for (TActorIterator<AActor> ActorIt(GetWorld()); ActorIt; ++ActorIt)
	{
		AActor* Actor = *ActorIt;
		if (Actor)
		{
			UHybridWaterPlaneComponent* WaterComp = Actor->FindComponentByClass<UHybridWaterPlaneComponent>();
			if (WaterComp)
			{
				CachedWaterSystem = WaterComp;
				UE_LOG(LogTemp, Log, TEXT("Chunk: Found water system"));
				return;
			}
		}
	}

	UE_LOG(LogTemp, Warning, TEXT("Chunk: No water system found in level"));
}

bool ADestructibleChunkActor::IsChunkInWater() const
{
	/**
	 * CHECK IF CHUNK IS IN WATER
	 * 
	 * Called each frame by Tick
	 * Determines if water forces should apply
	 * 
	 * ALGORITHM:
	 * 1. Check if water system exists
	 * 2. Query water height at chunk location
	 * 3. Compare to chunk Z position
	 * 4. If chunk Z < water Z: In water
	 * 
	 * PERFORMANCE:
	 * ~0.02ms per check
	 * Safe to call every frame
	 * 
	 * RETURN:
	 * true = chunk in water (apply forces)
	 * false = chunk above water (no forces)
	 */

	if (!CachedWaterSystem)
	{
		return false;
	}

	float WaterHeight = CachedWaterSystem->GetWaterHeightAt(GetActorLocation());
	return GetActorLocation().Z < WaterHeight;
}

float ADestructibleChunkActor::GetWaterHeightAtChunk() const
{
	/**
	 * QUERY WATER SURFACE HEIGHT
	 * 
	 * Used for buoyancy calculations
	 * Tells how deep chunk is in water
	 * 
	 * RETURN:
	 * Z coordinate of water surface at chunk
	 * 0 if no water system
	 */

	if (!CachedWaterSystem)
	{
		return 0.0f;
	}

	return CachedWaterSystem->GetWaterHeightAt(GetActorLocation());
}

void ADestructibleChunkActor::ApplyWaterForces()
{
	/**
	 * APPLY WATER PHYSICS
	 * 
	 * Called each frame if chunk in water
	 * Applies:
	 * - Buoyancy (upward force)
	 * - Drag (resistance to movement)
	 * - Current forces (from water flow)
	 * 
	 * BUOYANCY:
	 * F_buoyancy = Buoyancy * Volume * Gravity
	 * 0 = chunk sinks
	 * 1 = neutral (floats at surface)
	 * >1 = bobs up
	 * 
	 * DRAG:
	 * F_drag = -Velocity * DragCoefficient
	 * Slows down underwater movement
	 * More realistic water
	 * 
	 * CURRENT:
	 * F_current = WaterVelocity * Strength
	 * Pushes chunk along with water flow
	 * Creates interaction with waves/currents
	 */

	if (!ChunkMesh || !ChunkMesh->IsSimulatingPhysics())
	{
		return;
	}

	// ========================================================================
	// BUOYANCY FORCE
	// ========================================================================

	/**
	 * UPWARD FORCE WHEN SUBMERGED
	 * 
	 * Simulates water displacement
	 * Prevents chunks sinking instantly
	 * Creates floating effect
	 * 
	 * CALCULATION:
	 * Force = Buoyancy * MassMultiplier * Gravity
	 * Applied upward
	 * 
	 * DESIGNER CONTROL:
	 * 0.0 = no buoyancy (sinks fast)
	 * 0.5 = some buoyancy (slow sink)
	 * 1.0 = neutral buoyancy (floats)
	 * 1.5 = extra buoyancy (bobs)
	 */

	if (Buoyancy > 0.0f)
	{
		float ChunkMass = ChunkMesh->GetMass();
		float BuoyancyForce = Buoyancy * ChunkMass * 981.0f;  // 981 cm/s² gravity in UE
		ChunkMesh->AddForce(FVector::UpVector * BuoyancyForce);
	}

	// ========================================================================
	// DRAG FORCE
	// ========================================================================

	/**
	 * WATER RESISTANCE
	 * 
	 * Slows chunk movement through water
	 * Creates viscous effect
	 * 
	 * F_drag = -Velocity * DragCoefficient
	 * Opposite to velocity direction
	 * Magnitude increases with speed
	 * 
	 * DESIGNER CONTROL:
	 * 0.5 = low drag (slippery water)
	 * 1.0 = normal drag
	 * 2.0 = high drag (thick water)
	 */

	FVector ChunkVelocity = ChunkMesh->GetPhysicsLinearVelocity();
	if (!ChunkVelocity.IsNearlyZero())
	{
		FVector DragForce = -ChunkVelocity * WaterDragCoefficient;
		ChunkMesh->AddForce(DragForce);
	}

	// ========================================================================
	// WATER CURRENT FORCE
	// ========================================================================

	/**
	 * PUSH FROM WATER FLOW
	 * 
	 * Water surface has currents (from waves, wind)
	 * Chunks experience this force
	 * Gets pushed along with flow
	 * 
	 * QUERY:
	 * Get water velocity at chunk location
	 * This includes all water physics
	 * FFT waves, SWE ripples, etc.
	 * 
	 * FORCE:
	 * Scaled by submerged volume
	 * Deeper chunks get more force
	 * Surface chunks barely affected
	 */

	if (CachedWaterSystem)
	{
		FVector WaterVelocity = CachedWaterSystem->GetVelocityAt(GetActorLocation());
		if (!WaterVelocity.IsNearlyZero())
		{
			// Scale by estimated submerged fraction
			float SubmergedFraction = 0.5f;  // Estimate: half-submerged
			FVector CurrentForce = WaterVelocity * SubmergedFraction * ChunkMesh->GetMass();
			ChunkMesh->AddForce(CurrentForce);
		}
	}
}

// ============================================================================
// COLLISION & IMPACT EFFECTS
// ============================================================================

void ADestructibleChunkActor::OnChunkHit(UPrimitiveComponent* HitComponent, AActor* OtherActor,
	UPrimitiveComponent* OtherComp, FVector NormalImpulse, const FHitResult& Hit)
{
	/**
	 * COLLISION CALLBACK
	 * 
	 * Called when chunk hits surface
	 * Spawns impact effects (VFX, SFX)
	 * 
	 * TRIGGERED BY:
	 * - Chunk hits ground
	 * - Chunk hits wall
	 * - Chunk hits water
	 * - Chunk hits another object
	 * 
	 * FEEDBACK:
	 * Visual: Particle explosion, dust puff
	 * Audio: Impact sound, material-specific
	 * 
	 * COOLDOWN:
	 * Prevents sound spam
	 * Multiple impacts per frame = one sound
	 * 
	 * DESIGNER CONTROL:
	 * bEnableImpactFeedback toggle
	 * Disable for performance if needed
	 */

	if (!bEnableImpactFeedback)
	{
		return;
	}

	// ========================================================================
	// COOLDOWN CHECK (PREVENT SOUND SPAM)
	// ========================================================================

	/**
	 * RAPID COLLISION AVOIDANCE
	 * 
	 * Chunks can hit multiple times per frame
	 * (bouncing, tumbling, etc.)
	 * 
	 * Playing sound every hit = ear rape
	 * Implement cooldown between impacts
	 * 
	 * ImpactSoundCooldown = 0.1 seconds (default)
	 * After impact, wait 0.1s before next sound
	 * 
	 * CHECK:
	 * Current time - LastImpactTime < cooldown?
	 * If yes: Skip this impact sound
	 * If no: Play sound and update LastImpactTime
	 */

	if (GetWorld())
	{
		float TimeSinceLastImpact = GetWorld()->GetTimeSeconds() - LastImpactTime;
		if (TimeSinceLastImpact < ImpactSoundCooldown)
		{
			return;  // Cooldown active, skip sound
		}
		LastImpactTime = GetWorld()->GetTimeSeconds();
	}

	// ========================================================================
	// SPAWN IMPACT PARTICLE EFFECT
	// ========================================================================

	/**
	 * VISUAL FEEDBACK
	 * 
	 * Particle effect on impact
	 * Examples:
	 * - Dust puff (sand/dirt ground)
	 * - Spark burst (metal surface)
	 * - Splash (water surface)
	 * - Smoke (hard impact)
	 * 
	 * SPAWNING:
	 * One-time effect at impact location
	 * Plays immediately
	 * Removes itself when complete
	 * 
	 * DESIGNER NOTE:
	 * Leave ImpactParticleSystem empty for no effect
	 * Reduces complexity if not needed
	 */

	if (ImpactParticleSystem)
	{
		UNiagaraFunctionLibrary::SpawnSystemAtLocation(
			GetWorld(),
			ImpactParticleSystem,
			Hit.ImpactPoint,  // Where chunk hit
			Hit.ImpactNormal.Rotation()  // Align with surface
		);
	}

	// ========================================================================
	// PLAY IMPACT SOUND EFFECT
	// ========================================================================

	/**
	 * AUDIO FEEDBACK
	 * 
	 * Sound effect on impact
	 * Provides immediate audio response
	 * 
	 * EXAMPLES:
	 * - "clink" for metal
	 * - "thud" for wood
	 * - "splash" for water
	 * - "crack" for glass
	 * 
	 * FUTURE ENHANCEMENT:
	 * Could use surface type to select sound
	 * Different sounds per material
	 * More immersive
	 * 
	 * VOLUME:
	 * Scaled by impact force
	 * Harder impacts = louder sounds
	 * 
	 * CURRENT:
	 * Simple single sound
	 * Plays at impact location
	 */

	if (ImpactSoundEffect)
	{
		UGameplayStatics::PlaySoundAtLocation(
			GetWorld(),
			ImpactSoundEffect,
			Hit.ImpactPoint,
			1.0f  // Volume
		);
	}

	UE_LOG(LogTemp, Log, TEXT("Chunk impact: %s at %s"), *GetName(), *Hit.ImpactPoint.ToString());
}

// ============================================================================
// LIFECYCLE & POOLING
// ============================================================================

void ADestructibleChunkActor::DestroyChunk()
{
	/**
	 * DESTROY CHUNK
	 * 
	 * Called when:
	 * - Lifetime expires
	 * - Manual call
	 * - Pooling system returns to pool
	 * 
	 * CLEANUP SEQUENCE:
	 * 1. Unregister from wind system
	 * 2. Hide actor
	 * 3. Disable collision
	 * 4. Destroy or return to pool
	 * 
	 * POOLING VS DESTRUCTION:
	 * Pooling: Actor reused (better performance)
	 * Destruction: Actor deleted (cleaner)
	 * 
	 * DESIGNER DECISION:
	 * Parent destructible controls pooling
	 * This just provides interface
	 */

	if (!bIsActive)
	{
		return;  // Already destroyed
	}

	// Unregister systems
	UnregisterFromWindSystem();

	// Hide and disable
	SetActorHiddenInGame(true);
	SetActorEnableCollision(false);
	ChunkMesh->SetSimulatePhysics(false);

	// Mark as inactive (for pooling)
	bIsActive = false;

	UE_LOG(LogTemp, Log, TEXT("Chunk destroyed: %s"), *GetName());
}

void ADestructibleChunkActor::ResetChunk()
{
	/**
	 * RESET FOR POOLING
	 * 
	 * Called when chunk returned to pool
	 * Clears all physics state
	 * Prepares for reuse
	 * 
	 * RESET:
	 * - Clear velocities
	 * - Reset position/rotation
	 * - Stop particles
	 * - Reset timers
	 * 
	 * AFTER RESET:
	 * Chunk ready to spawn again
	 * Same object, fresh state
	 * No allocation overhead
	 */

	if (!ChunkMesh)
	{
		return;
	}

	// Clear movement
	ChunkMesh->SetPhysicsLinearVelocity(FVector::ZeroVector);
	ChunkMesh->SetPhysicsAngularVelocityInRadians(FVector::ZeroVector);

	// Reset particles
	if (TrailParticles)
	{
		TrailParticles->ResetSystem();
	}

	// Reset timers
	TimeAlive = 0.0f;
	LastImpactTime = -1.0f;

	// Make visible but disabled
	SetActorHiddenInGame(true);
	SetActorEnableCollision(false);

	UE_LOG(LogTemp, Log, TEXT("Chunk reset: %s"), *GetName());
}

bool ADestructibleChunkActor::IsChunkActive() const
{
	/**
	 * QUERY ACTIVE STATUS
	 * 
	 * Used by pooling system
	 * Determines if chunk available for reuse
	 * 
	 * RETURN:
	 * true = chunk currently active
	 * false = chunk available for pooling
	 */

	return bIsActive;
}

// ============================================================================
// FACTORY METHOD
// ============================================================================

ADestructibleChunkActor* ADestructibleChunkActor::SpawnChunkAtLocation(
	UWorld* World,
	UStaticMesh* Mesh,
	const FVector& Location,
	const FRotator& Rotation,
	const FVector& ImpulseDir,
	float ImpulseMag)
{
	/**
	 * SPAWN CHUNK FACTORY METHOD
	 * 
	 * Static function for convenient chunk creation
	 * Called from parent destructible
	 * 
	 * PROCESS:
	 * 1. Create actor in world
	 * 2. Set mesh and physics data
	 * 3. Return configured chunk
	 * 
	 * EXAMPLE USAGE:
	 * for (int32 i = 0; i < ChunkCount; i++)
	 * {
	 *     ADestructibleChunkActor* Chunk = ADestructibleChunkActor::SpawnChunkAtLocation(
	 *         GetWorld(),
	 *         ChunkMeshes[i % ChunkMeshes.Num()],
	 *         GetActorLocation(),
	 *         FRotator::ZeroRotator,
	 *         (FMath::VRand() * FVector(1,1,0.5f)).GetSafeNormal(),
	 *         ChunkImpulseStrength
	 *     );
	 * }
	 * 
	 * @param World World to spawn in
	 * @param Mesh Static mesh asset
	 * @param Location Spawn position (world space)
	 * @param Rotation Initial rotation
	 * @param ImpulseDir Force direction (normalized)
	 * @param ImpulseMag Force magnitude (Newtons)
	 * @return Newly created chunk actor
	 */

	if (!World || !Mesh)
	{
		return nullptr;
	}

	// Spawn actor
	ADestructibleChunkActor* Chunk = World->SpawnActor<ADestructibleChunkActor>(
		ADestructibleChunkActor::StaticClass(),
		Location,
		Rotation
	);

	if (!Chunk)
	{
		return nullptr;
	}

	// Configure chunk
	Chunk->ChunkStaticMesh = Mesh;
	Chunk->PhysicsData.ImpulseDirection = ImpulseDir.GetSafeNormal();
	Chunk->PhysicsData.ImpulseMagnitude = ImpulseMag;
	Chunk->PhysicsData.AngularImpulse = FMath::FRandRange(300.0f, 1000.0f);  // Random spin
	Chunk->SpawnLocation = Location;
	Chunk->SpawnRotation = Rotation;

	return Chunk;
}

	 * Small object (barrel, box): 3-5 chunks
	 * Medium object (crate, furniture): 5-8 chunks
	 * Large object (building piece): 8-15 chunks

	 * 
	 * PERFORMANCE CONSIDERATION:
	 * Each chunk: ~0.5ms physics per frame

	 * 
	 * RECOMMENDATION:
	 * Start with 5 chunks

	 * Designer Notes:

	 * 
	 * PHYSICS:
	 * Applied as Newton impulse units
	 * Roughly: 100 N = chunk flies ~1 meter/second
	 * (exact depends on chunk mass)
	 * 
	 * GAMEPLAY TUNING:
	 * Weak explosions: 200-500 N (objects barely move)
	 * Normal explosions: 500-1500 N (standard destruction)
	 * Powerful explosions: 1500-3000 N (dramatic destruction)
	 * Extreme explosions: 3000+ N (objects fly far)
	 * 
	 * GAME FEEL:
	 * Player preference dictates range
	 * Test and adjust until feels right
	 * 
	 * VISUAL REFERENCE:
	 * Ratchet & Clank: ~1000-2000 N range
	 * Destroyer: ~500-1000 N range (more realistic)
	 * Bayonetta: ~2000-4000 N range (over-the-top)
	 * 
	 * WIND INTERACTION:
	 * Faster chunks blown more by wind
	 * Slower chunks less affected
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Destructible|Chunks",
		meta = (UIMin = "0", UIMax = "5000", ClampMin = "0", ClampMax = "20000",
		Tooltip = "Explosion impulse (Newtons). 500=normal, 2000=dramatic"))
	float ChunkImpulseStrength = 1000.0f;

	/**
	 * CHUNK LIFETIME (Auto-Despawn Time)
	 * 
	 * Designer Notes:
	 * How long chunks persist after spawn
	 * Then automatically removed from world
	 * 
	 * GAMEPLAY BALANCE:
	 * Too short (0.5s): Chunks disappear too fast (looks cheap)
	 * Normal (3s): Chunks visible for reasonable time
	 * Too long (10s+): Chunks clutter level, performance drops
	 * 
	 * DESIGNER RECOMMENDATIONS:
	 * Outdoor levels: 3-5 seconds (lots of space)
	 * Indoor levels: 2-3 seconds (less space)
	 * Combat arena: 1-2 seconds (multiple destructions)
	 * 
	 * IMPLEMENTATION:
	 * Each chunk tracks elapsed time
	 * When time > lifetime: chunk returns to pool
	 * Returns chunks for reuse (performance)
	 * 
	 * ZERO = INFINITE:
	 * If set to 0: Chunks persist indefinitely
	 * Must manually cleanup
	 * Not recommended for performance
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Destructible|Chunks",
		meta = (UIMin = "0.5", UIMax = "10", ClampMin = "0",
		Tooltip = "How long chunks stay before despawn (seconds)"))
	float ChunkLifetime = 3.0f;

	// ========================================================================
	// CHUNK PHYSICS CONFIGURATION
	// ========================================================================

	/**
	 * CHUNK MASS MULTIPLIER
	 * 
	 * Designer Notes:
	 * Controls how heavy chunks are
	 * Affects wind interaction significantly
	 * 
	 * SCALE:

	float ChunkMassMultiplier = 1.0f;

	 * CHUNK SPIN (Angular Impulse)
	 * 
	 * Designer Notes:
	 * How much chunks spin after spawn
	 * Radians per second
	 * 
	 * VISUAL EFFECT:
	 * 0 = no spin (boring, static)
	 * 300 = gentle rotation (subtle)
	 * 1000 = dramatic spinning (dynamic)
	 * 3000+ = extreme tumbling (over-the-top)
	 * 
	 * RECOMMENDATION:
	 * Random between 300-1000 for variety

	 * 
	 * REALISM VS DRAMA:
	 * Realistic: 300-500 rad/sec
	 * Stylized: 1000-2000 rad/sec
	 * Arcade: 2000+ rad/sec
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Destructible|Chunks|Physics",

	float ChunkSpinStrength = 500.0f;

	/**
	 * CHUNK WIND RESISTANCE
	 * 
	 * Designer Notes:
	 * How much wind affects chunks

	 * 
	 * SCALE:
	 * 0.1 = very resistant (barely moves)
	 * 0.5 = medium resistant
	 * 1.0 = normal (equal to wind effect)

	 * 
	 * COMBINED WITH MASS:
	 * Heavy chunks + low resistance = slow movement
	 * Light chunks + high resistance = fast movement
	 * 
	 * TUNING:
	 * Metal debris: 0.3 (heavy, resistant)
	 * Paper/cloth: 2.0 (light, susceptible)
	 * Normal: 1.0
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Destructible|Chunks|Physics",
		meta = (UIMin = "0.1", UIMax = "3", ClampMin = "0.01",
		Tooltip = "Wind resistance (1=normal, 0.5=heavy, 2=light)"))
	float ChunkWindResistance = 1.0f;

	// ========================================================================
	// WATER INTERACTION
	// ========================================================================

	/**
	 * WATER INTERACTION TYPE
	 * 
	 * Designer Notes:
	 * How chunks behave in water
	 * Sink, Float, Splash, or None
	 * 
	 * SINK:
	 * Chunks immediately fall through water
	 * No buoyancy
	 * Quickly disappear underwater
	 * 
	 * FLOAT:
	 * Chunks bob at water surface
	 * Pushed by currents
	 * Can be blown by wind
	 * Realistic for wood, foam
	 * 
	 * SPLASH:
	 * Chunks hit water and splash
	 * Then continue falling
	 * Sound/particle effects
	 * 
	 * NONE:
	 * No water interaction at all
	 * Useful for gameplay balance
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Destructible|Chunks|Water",
		meta = (Tooltip = "How chunks interact with water"))
	UENUM(BlueprintType)
	enum class EChunkWaterInteraction : uint8
	{
		Sink UMETA(DisplayName = "Sink"),
		Float UMETA(DisplayName = "Float"),
		Splash UMETA(DisplayName = "Splash"),
		None UMETA(DisplayName = "No Interaction")
	} ChunkWaterInteraction = EChunkWaterInteraction::Splash;

	/**
	 * CHUNK BUOYANCY (Floating)
	 * 
	 * Designer Notes:
	 * How much chunks float in water
	 * Only used if WaterInteraction = Float
	 * 
	 * SCALE:
	 * 0.0 = sinks (no buoyancy)
	 * 0.5 = slow sink (some buoyancy)
	 * 1.0 = neutral (floats at surface)
	 * 1.5 = bobs up (extra buoyant)
	 * 
	 * PHYSICS:
	 * F_buoyancy = buoyancy × mass × gravity
	 * Higher = more upward force
	 * 
	 * GAMEPLAY:
	 * Wood barrel: 1.0 (floats)
	 * Metal barrel: 0.2 (sinks slowly)
	 * Cork chunk: 1.5 (bobs energetically)
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Destructible|Chunks|Water",
		meta = (UIMin = "0", UIMax = "2", ClampMin = "0",
		Tooltip = "Buoyancy in water (1=floats, 0=sinks)"))
	float ChunkBuoyancy = 0.5f;

	/**
	 * WATER DRAG
	 * 
	 * Designer Notes:
	 * Resistance to movement through water
	 * Makes underwater motion slower
	 * 
	 * SCALE:
	 * 0.5 = low drag (slippery)
	 * 1.0 = normal drag
	 * 2.0 = high drag (thick water)
	 * 
	 * EFFECT:
	 * Multiplies velocity to get drag force
	 * Higher = slower underwater movement
	 * 
	 * TUNING:
	 * Realistic water: 1.0-1.5
	 * Game feel: Adjust as needed
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Destructible|Chunks|Water",
		meta = (UIMin = "0.5", UIMax = "3", ClampMin = "0.1",
		Tooltip = "Water resistance (1=normal, 2=thick water)"))
	float ChunkWaterDrag = 1.0f;

	// ========================================================================
	// VISUAL FEEDBACK
	// ========================================================================

	/**
	 * TRAIL PARTICLE SYSTEM
	 * 
	 * Designer Notes:
	 * Particles that follow chunks
	 * Examples: Dust trail, fire trail, magic trail
	 * 
	 * ASSIGNMENT:
	 * Select Niagara system asset
	 * Will spawn on every chunk
	 * Attaches to chunk, follows movement
	 * 
	 * LEAVE EMPTY:
	 * For no trail effect
	 * Reduces particle count
	 * Better performance
	 * 
	 * EXAMPLES:
	 * Wood crate: Dust trail
	 * Metal object: Spark trail
	 * Magical object: Magic particle trail
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Destructible|Feedback",
		meta = (Tooltip = "Particle system following chunks (optional)"))
	class UNiagaraSystem* ChunkTrailParticles;

	/**
	 * IMPACT PARTICLE SYSTEM
	 * 
	 * Designer Notes:
	 * Particles spawned when chunk hits surface
	 * One-time effect
	 * 
	 * EXAMPLES:
	 * Dust puff (ground hit)
	 * Spark burst (metal impact)
	 * Water splash (water hit)
	 * 
	 * GAMEPLAY:
	 * Provides feedback when chunks land
	 * Makes destruction feel impactful
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Destructible|Feedback",
		meta = (Tooltip = "Particle effect when chunk impacts (optional)"))
	class UNiagaraSystem* ImpactParticles;

	/**
	 * IMPACT SOUND
	 * 
	 * Designer Notes:
	 * Sound played when chunk hits surface
	 * 
	 * ASSIGNMENT:
	 * Assign sound effect
	 * Will play on all chunk impacts
	 * Volume auto-adjusted by engine
	 * 
	 * FUTURE ENHANCEMENT:
	 * Could use surface type to select sound
	 * Different sounds for wood, metal, glass
	 * More immersive audio feedback
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Destructible|Feedback",
		meta = (Tooltip = "Sound when chunk impacts (optional)"))
	class USoundBase* ImpactSound;

	/**
	 * DESTRUCTION VFX
	 * 
	 * Designer Notes:
	 * Major explosion effect when object destroyed
	 * Spawns at destruction point
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Destructible|Feedback",
		meta = (Tooltip = "Explosion effect when destroyed (optional)"))
	class UNiagaraSystem* DestructionVFX;

	/**
	 * DESTRUCTION SOUND
	 * 
	 * Designer Notes:
	 * Explosion sound when object destroyed
	 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Destructible|Feedback",
		meta = (Tooltip = "Explosion sound when destroyed (optional)"))
	class USoundBase* DestructionSFX;

	// ========================================================================
	// PUBLIC FUNCTIONS
	// ========================================================================

	/**
	 * TAKE DAMAGE
	 * 
	 * Blueprint Callable
	 * 
	 * Called when object hit/attacked
	 * Reduces health
	 * Triggers destruction at 0 health
	 */
	UFUNCTION(BlueprintCallable, Category = "Destructible",
		meta = (Tooltip = "Apply damage to destructible"))
	virtual void TakeDamageCustom(float DamageAmount, AActor* DamageCauser);

	/**
	 * DESTROY OBJECT
	 * 
	 * Blueprint Callable
	 * 
	 * Triggers destruction immediately
	 * Spawns chunks, VFX, SFX
	 * Removes intact mesh
	 */
	UFUNCTION(BlueprintCallable, Category = "Destructible",
		meta = (Tooltip = "Destroy object and spawn chunks"))
	virtual void DestroyObject();

protected:

	virtual void BeginPlay() override;
	virtual void Tick(float DeltaTime) override;

	// ========================================================================
	// CHUNK SPAWNING
	// ========================================================================

	/**
	 * SPAWN ALL CHUNKS
	 * 
	 * Internal function
	 * Called by DestroyObject
	 * Creates configured number of chunks
	 */
	UFUNCTION(BlueprintCallable, Category = "Destructible")
	void SpawnChunks();

	/**
	 * SPAWN SINGLE CHUNK
	 * 
	 * Helper function
	 * Creates one chunk with configuration
	 */
	UFUNCTION(BlueprintCallable, Category = "Destructible")
	ADestructibleChunkActor* SpawnSingleChunk();

	// ========================================================================
	// DATA REWARDS
	// ========================================================================

	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Destructible|Rewards",
		meta = (ClampMin = "0",
		Tooltip = "Data currency to drop on destruction"))
	int32 DataDropAmount = 10;

	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Destructible|Rewards",
		meta = (Tooltip = "Auto-collect data within radius (no pickup spawned)"))
	bool bAutoCollectData = true;

	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Destructible|Rewards",
		meta = (UIMin = "0", UIMax = "1000", ClampMin = "0",
		Tooltip = "Radius for auto-collection (cm)"))
	float AutoCollectRadius = 200.0f;

	// ========================================================================
	// INTERNAL STATE
	// ========================================================================

	UPROPERTY()
	class UStaticMeshComponent* IntactMesh;

	UPROPERTY()
	TArray<ADestructibleChunkActor*> ActiveChunks;
};
