/**
 * ============================================================================
 * HYBRID WATER SYSTEM - HEADER FILE
 * ============================================================================
 * 
 * OVERVIEW:
 * This system provides realistic water simulation using a three-layer approach:
 * 
 * 1. FFT OCEAN WAVES (Layer 1 - Visual Foundation)
 *    - Runs on GPU using Fast Fourier Transform
 *    - Generates realistic ocean-scale waves based on wind parameters
 *    - Always active, minimal performance cost (~1ms)
 *    - Perfect for distant water, large bodies, and background waves
 * 
 * 2. SHALLOW WATER EQUATIONS (Layer 2 - Interactive Physics)
 *    - Real-time fluid dynamics simulation
 *    - Responds to actor impacts and movement
 *    - GPU compute shader-based, stable up to 256x256 grid
 *    - Creates realistic ripples and wave propagation (~3.5ms)
 * 
 * 3. SPH PARTICLES (Layer 3 - High-Fidelity Local)
 *    - Smoothed Particle Hydrodynamics for actor interactions
 *    - Creates particle splashes when actors move through water
 *    - Limited to 256 particles for real-time performance
 *    - Best for camera-close interactions (~1.5ms)
 * 
 * DESIGN PHILOSOPHY:
 * - Smart LOD: Only compute what's visible and important
 * - Layered approach: Each layer can be independently adjusted
 * - Performance-first: Target 60 FPS on all platforms
 * - Realism: Physical accuracy within real-time constraints
 * 
 * PERFORMANCE TARGET: 7-8ms per frame (60 FPS)
 * - Allows ~2ms buffer for other game logic
 * - Scales gracefully on lower-end hardware
 * 
 * ============================================================================
 */

/**
 * ============================================================================
 * LAYER 1: FFT OCEAN WAVES - STRUCT & CLASS
 * ============================================================================
 * 
 * FFT (Fast Fourier Transform) generates realistic ocean waves by:
 * 1. Computing Phillips spectrum (wind → wave energy distribution)
 * 2. Performing FFT to convert frequency domain → spatial domain
 * 3. Computing normal maps for realistic lighting
 * 
 * DESIGNER NOTES:
 * - Adjust WindSpeed and WindDirection to change wave character
 * - Amplitude controls wave height (1.0 = real-world scale)
 * - ChoppinessScale affects Gerstner wave distortion (visual chop)
 * - These parameters directly affect player immersion
 * 
 * PROGRAMMER NOTES:
 * - This is a render resource (GPU-allocated memory)
 * - Always runs; use LOD/culling at usage point if needed
 * - Update frequency: Once per frame (~60 Hz)
 * - Thread-safe by design (GPU-side computation)
 */

struct FFFTWaveParameters
{
	/** Wave height multiplier (1.0 = ~0.5m waves, 2.0 = ~1m waves) */
	float Amplitude = 1.0f;

	/** Primary wind direction (normalized). Affects wave propagation direction */
	FVector2D WindDirection = FVector2D(1.0f, 0.0f);

	/** Wind speed in m/s. Higher values create larger, more energetic waves */
	float WindSpeed = 10.0f;

	/** Gerstner wave choppiness multiplier. Controls visual wave distortion (0.0 = flat, 2.0 = very choppy) */
	float ChoppinessScale = 1.0f;

	/** Simulation time multiplier. 1.0 = real-time, 2.0 = fast motion, 0.5 = slow motion */
	float TimeScale = 1.0f;
};

/**

/**
 * ============================================================================
 * LAYER 2: SHALLOW WATER EQUATIONS - STRUCT & CLASS
 * ============================================================================
 * 
 * Shallow Water Equations (SWE) simulate real fluid dynamics on a 2D grid.
 * Perfect for interactive ripples, waves, and realistic water behavior.
 * 
 * Physics Equations:
 *   dh/dt = -(d(hu)/dx + d(hv)/dy)           [Height continuity]
 *   d(hu)/dt = -d(hu²/h)/dx - g·h·dh/dx      [X-momentum]
 *   d(hv)/dt = -d(hv²/h)/dy - g·h·dh/dy      [Y-momentum]
 * 
 * Where: h=height, u/v=velocities, g=gravity
 * 
 * DESIGNER NOTES:
 *   - GridSize controls simulation resolution vs quality trade-off
 *   - Smaller cells = more detail but slower (quadratic cost)
 *   - Damping controls energy dissipation (1.0 = forever, 0.9 = fades quickly)
 *   - Impact force directly controls ripple intensity
 * 
 * PROGRAMMER NOTES:
 *   - Runs on GPU via compute shader for parallel computation
 *   - CFL condition ensures stability (time step must be small enough)
 *   - Blended with FFT layer for seamless visual result
 *   - Compute budget: ~3-5ms for 256x256 grid
 */

struct FSWEGridCell
{
	float Height;           // Water height above base plane (cm)
	float VelocityX;        // Horizontal velocity component (cm/s)
	float VelocityY;        // Vertical velocity component (cm/s)
	float NextHeight;       // Temporary storage for next frame calculation
	float NextVelocityX;    // Temporary storage for next frame calculation
	float NextVelocityY;    // Temporary storage for next frame calculation
};

/**
 * FShallowWaterSimulation
 * 
 * Real-time fluid dynamics using Shallow Water Equations on GPU.
 * Simulates interactive water response to actor movement and impacts.
 * 
 * DESIGNER: Key tuning parameters for water feel:
 *   - GridWidth/GridHeight: 128x128 (fast, decent detail)
 *                          256x256 (default, balanced)
 *                          512x512 (slow, very detailed)
 *   - CellSize: Larger = coarser grid, smaller = finer detail
 *   - Damping: How quickly waves fade (0.99 = very persistent, 0.95 = quick fade)
 * 
 * PROGRAMMER: Compute shader based for GPU parallelism.
 *   - Each thread handles one grid cell
 *   - Double-buffering prevents read-write conflicts
 *   - Requires RHI command list access for dispatch
 */
class FShallowWaterSimulation : public FRenderResource
{
public:
	/**
	 * Constructor
	 * @param InGridWidth Number of cells in X direction (power of 2 preferred: 128, 256, 512)
	 * @param InGridHeight Number of cells in Y direction (power of 2 preferred)
	 * @param InCellSize Physical size of each cell in Unreal units (100 UU typical)
	 * 
	 * DESIGNER: Total simulation area = GridWidth * GridHeight * CellSize
	 *   Example: 256x256 grid with 100 UU cells = 25,600 x 25,600 UU area
	 */
	FShallowWaterSimulation(int32 InGridWidth, int32 InGridHeight, float InCellSize);
	virtual ~FShallowWaterSimulation();

	// FRenderResource Interface
	virtual void InitializeRHI(FRHICommandListBase& RHICmdList) override;
	virtual void ReleaseRHI() override;

	/**
	 * Simulate one frame of shallow water physics
	 * CALL THIS: Once per frame, but only if camera is within SWESimulationDistance
	 * 
	 * @param DeltaTime Frame time. Function will enforce CFL stability internally.
	 * 
	 * DESIGNER: Don't call every frame if water is offscreen for performance.
	 * PROGRAMMER: CFL condition automatically reduces internal timestep if needed.
	 */
	void UpdateSimulation(float DeltaTime);

	/**
	 * Apply an impact (splash) to the water
	 * Creates a height perturbation that propagates as waves.
	 * 
	 * @param ImpactLocation World position of impact center
	 * @param ImpactForce How hard the impact is (1000 = strong splash)
	 * @param ImpactRadius How large the splash area is in UU (100-500 typical)
	 * 
	 * DESIGNER: Use this for:
	 *   - Character jumping: Force=ImpactVelocity*Mass, Radius=CharacterRadius*2
	 *   - Projectile splash: Force=ProjectileDamage, Radius=ProjectileSize
	 *   - Rain: Force=10-50, Radius=50 (scattered across surface)
	 * 
	 * PROGRAMMER: Converts world coords to grid coords, applies impulse.
	 */
	void ApplyImpact(const FVector& ImpactLocation, float ImpactForce, float ImpactRadius);
	
	// Query functions for physics integration
	float GetHeightAt(const FVector& WorldLocation) const;
	FVector GetVelocityAt(const FVector& WorldLocation) const;
	
	// Render resource access
	FRHITexture* GetHeightField() const { return HeightFieldTexture.GetReference(); }

private:
	int32 GridWidth, GridHeight;
	float CellSize;
	float Gravity = 9.81f;           // m/s² (standard Earth gravity)
	float Damping = 0.99f;           // Energy dissipation per frame

	// GPU Textures - double-buffered for read/write safety
	TRefCountPtr<FRHITexture2D> HeightFieldTexture;
	TRefCountPtr<FRHITexture2D> VelocityFieldTexture;
	TRefCountPtr<FRHITexture2D> HeightFieldTemp;
	TRefCountPtr<FRHITexture2D> VelocityFieldTemp;

	// Compute kernel dispatches (called by UpdateSimulation)
	void ComputeHeightContinuity(float DeltaTime);    // dh/dt = -div(hu, hv)
	void ComputeMomentumEquations(float DeltaTime);   // du/dt = -grad(u²/2 + gh)
	void EnforceCFLStability(float& DeltaTime);       // Ensure dt < cell_size/c
};

/**
 * ============================================================================
 * LAYER 3: SPH PARTICLES - STRUCT & CLASS
 * ============================================================================
 * 
 * Smoothed Particle Hydrodynamics (SPH) simulates water as discrete particles.
 * Creates realistic splashes and spray effects when actors interact with water.
 * 
 * Key Concepts:
 *   - Particles represent water volume (not individual water molecules)
 *   - Each particle has density, pressure, velocity
 *   - Forces propagate through nearby particles via smoothing kernel
 *   - Perfect for close-range, high-fidelity interactions
 * 
 * DESIGNER NOTES:
 *   - Visible impact on screen: noticeable splashes, foam, spray
 *   - Limited to ~256 particles for 60 FPS on current hardware
 *   - Best when camera is close to water interaction
 *   - Combine with SWE for seamless effect (distant = SWE, close = SPH)
 * 
 * PROGRAMMER NOTES:
 *   - Quadratic complexity in particle count (neighbor search)
 *   - CPU-based for ease of debugging and tuning
 *   - Could be GPU-accelerated with spatial hashing if needed
 *   - Integrates with physics engine via force application
 */

struct FSPHParticle
{
	FVector Position;       // World space position (cm)
	FVector Velocity;       // Current velocity (cm/s)
	float Density;          // Current density (kg/m³)
	float Pressure;         // Computed pressure (Pa)
	float Mass = 0.001f;    // Particle mass (constant, ~1 gram)
};

/**
 * FLocalSPHInteraction
 * 
 * SPH particle simulation for close-range water interactions.
 * Generates visible splashes and spray effects.
 * 
 * DESIGNER: Tune these for desired splash effect:
 *   - SmoothingRadius: Larger = smoother, wider interaction (100 UU typical)
 *   - RestDensity: Reference density (1000 kg/m³ = water)
 *   - Viscosity: Thickness of water (0.5 = normal, 2.0 = thick/gooey)
 * 
 * PROGRAMMER: CPU-based for predictability. Key operations:
 *   - Find neighbors: O(n²) but limited to 256 particles
 *   - Compute density: kernel summation
 *   - Compute pressure: equation of state
 *   - Integrate: Euler method (simple, fast)
 */
class FLocalSPHInteraction
{
public:
	/**
	 * Constructor
	 * @param MaxParticles Maximum particles in simulation (256 recommended)
	 *                     More particles = more detail but slower
	 *                     256 particles = ~1.5ms on modern GPUs
	 */
	FLocalSPHInteraction(int32 MaxParticles = 256);
	~FLocalSPHInteraction();

	/** Initialize particle simulation */
	void Initialize();

	/**
	 * Update particle simulation one frame
	 * @param DeltaTime Frame time in seconds
	 * @param WaterPlaneCenter Center of water volume (for bounds checking)
	 */
	void UpdateParticles(float DeltaTime, const FVector& WaterPlaneCenter);

	/**
	 * Create splash at location
	 * @param ImpactLocation Where the splash occurs
	 * @param ImpactForce How hard the impact (affects particle count and velocity)
	 */
	void ApplyImpact(const FVector& ImpactLocation, float ImpactForce);

	/**
	 * Query interaction force at location (for actor physics integration)
	 * @param QueryLocation Where to sample force
	 * @param QueryRadius How far to search for particles
	 * @return Net force on an object at that location
	 */
	FVector GetInteractionForceAt(const FVector& QueryLocation, float QueryRadius) const;

	bool HasActiveParticles() const { return Particles.Num() > 0; }

	/** Add a particle emission source (splash point) */
	void AddParticleSource(const FVector& SourceLocation, const FVector& SourceVelocity, int32 ParticleCount);

private:
	TArray<FSPHParticle> Particles;
	int32 MaxParticles;
	
	// SPH Parameters (calibrated for water)
	static constexpr float SmoothingRadius = 100.0f;  // Interaction radius (cm)
	static constexpr float RestDensity = 1000.0f;      // Water density (kg/m³)
	static constexpr float Viscosity = 0.5f;           // Water viscosity

	// Neighbor search and force computation
	void FindNeighbors(int32 ParticleIndex, TArray<int32>& OutNeighbors);
	void ComputeDensity();                            // Kernel summation
	void ComputePressure();                           // Equation of state
	void ComputeForces();                             // Pressure & viscosity forces
	void Integrate(float DeltaTime);                  // Update positions/velocities
};

void FFFTOceanWaves::InitializeRHI(FRHICommandListBase& RHICmdList)
{
	/**
	 * Initialize GPU textures - Called on RENDER THREAD
	 * 
	 * FRenderResource PATTERN:
	 * 1. Game thread calls BeginInitResource()
	 * 2. Queues InitializeRHI() to execute on render thread
	 * 3. Render thread creates GPU resources safely
	 * 4. Game thread can access textures after initialization complete
	 * 
	 * TEXTURE FORMAT CHOICES:
	 * - PF_FloatRGBA: Provides high precision for wave data
	 *   R channel: Height displacement (critical precision)
	 *   G channel: Can store foam or other data
	 *   B/A: Future expansion (e.g., second wave set)
	 * 
	 * - PF_FloatRGBA vs PF_R32F?
	 *   RGBA takes more VRAM but allows multi-channel data
	 *   For 512x512: ~4MB per texture (acceptable)
	 * 
	 * TEXTURE FLAGS:
	 * - RenderTargetable: Can render TO this texture
	 * - ShaderResource: Can sample FROM this texture in shaders
	 * - UAV: Unordered Access View (compute shader writes)
	 * 
	 * WHY THREE TEXTURES?
	 * - HeightMap: Output of FFT (primary result)
	 * - NormalMap: Surface normals (for lighting)
	 * - FoamMap: Foam/spray intensity mask
	 * 
	 * DESIGNER IMPACT:
	 * Higher resolution = more wave detail but slower
	 * Most players won't notice above 512x512 unless inspecting close-up
	 */

	// Create height map texture
	// This stores the displacement of water surface
	FRHITextureCreateDesc Desc =
		FRHITextureCreateDesc::Create2D(TEXT("OceanHeightMap"))
		.SetExtent(Resolution, Resolution)                           // 512x512 or 1024x1024
		.SetFormat(PF_FloatRGBA)                                     // R32G32B32A32 float
		.SetFlags(ETextureCreateFlags::RenderTargetable |            // Can render to
				  ETextureCreateFlags::ShaderResource |              // Can sample in shaders
				  ETextureCreateFlags::UAV);                         // Can write via compute

	HeightMapTexture = RHICreateTexture(Desc);

	// Create normal map texture
	// Stores surface normals for realistic lighting
	// Computed as derivative of height map (dH/dX, dH/dY)
	NormalMapTexture = RHICreateTexture(Desc);

	// Create foam map texture
	// Indicates where foam/spray appears
	// Foam appears on wave peaks and steep slopes
	FoamMapTexture = RHICreateTexture(Desc);

	/**
	 * MEMORY FOOTPRINT:
	 * Each texture at 512x512:
	 * - 512 * 512 * 16 bytes (RGBA float) = 4 MB
	 * - Total 3 textures: 12 MB
	 * - Acceptable GPU overhead
	 * 
	 * At 1024x1024:
	 * - 1024 * 1024 * 16 bytes = 16 MB per texture
	 * - Total: 48 MB (still reasonable)
	 */

	// Initialize wave spectrum for first frame
	// Contains the mathematical basis for wave generation
	GeneratePhillipsSpectrum();
}

void FFFTOceanWaves::ReleaseRHI()
{
	/**
	 * Release GPU textures - Called on RENDER THREAD during shutdown
	 * 
	 * SAFE RELEASE PATTERN:
	 * SafeRelease() does:
	 * 1. Check if pointer is valid
	 * 2. Decrement reference count
	 * 3. GPU driver frees memory when refcount reaches 0
	 * 4. Set pointer to null
	 * 
	 * Why use SafeRelease?
	 * - Prevents double-delete crashes
	 * - Thread-safe (happens on render thread)
	 * - Reference counting handles multi-owner scenarios
	 * 
	 * CLEANUP ORDER:
	 * Order doesn't matter for these unrelated textures
	 * But in complex systems, release in reverse creation order
	 */

	HeightMapTexture.SafeRelease();
	NormalMapTexture.SafeRelease();
	FoamMapTexture.SafeRelease();

	/**
	 * AFTER THIS:
	 * - GPU memory freed
	 * - Pointers become null
	 * - Game thread should not access these
	 * - Component should be destroyed or recycled
	 */
}

void FFFTOceanWaves::PerformFFT()
{
	/**
	 * Fast Fourier Transform: Frequency Domain → Spatial Domain
	 * 
	 * WHAT IS FFT?
	 * Mathematical algorithm that converts data representation
	 * Input: Frequency domain (how much of each frequency)

	 * - Naive approach: O(N²) where N = grid size
	 * - FFT: O(N log N)

	 *   Naive: 68 billion operations
	 *   FFT: 4.7 million operations
	 *   SPEEDUP: ~14,500x faster!
	 * 
	 * WHY USE FFT FOR OCEAN?
	 * - Produces non-repetitive, complex wave patterns
	 * - Computationally efficient even on older hardware
	 * - GPU-friendly (highly parallel algorithm)
	 * 
	 * WAVE EQUATION SOLVED:
	 * Start: Wind spectrum (frequency domain)
	 * ↓
	 * Apply time-evolution (phase shifts)
	 * ↓
	 * FFT (transform)
	 * ↓
	 * End: Height field at current time (spatial domain)
	 * 
	 * DESIGNER IMPACT:
	 * Higher resolution FFT = more wave detail
	 * 512x512: Looks great, performance fine
	 * 1024x1024: Very detailed, but 2-3x slower
	 * 
	 * PROGRAMMER IMPLEMENTATION:

void FFFTOceanWaves::CalculateNormals()
{
	/**
	 * Calculate Surface Normal Map from Heights
	 * 
	 * WHY NORMALS?
	 * Lighting equation needs surface orientation
	 * Normals determine how light reflects off water
	 * Creates convincing shiny, specular appearance
	 * 
	 * HOW TO COMPUTE FROM HEIGHTS?
	 * Normal is perpendicular to surface
	 * Can be computed from height derivatives
	 * 
	 * Formula:
	 * normal = normalize(cross(dH/dx direction, dH/dy direction))
	 * 
	 * In 2D height field:
	 * slope_x = (H[i+1][j] - H[i-1][j]) / 2
	 * slope_y = (H[i][j+1] - H[i][j-1]) / 2
	 * normal = normalize((-slope_x, -slope_y, 1))
	 * 
	 * OPTIMIZATION: Precompute slope textures
	 * Store dH/dx in R channel
	 * Store dH/dy in G channel
	 * Reconstruct normal in shader
	 * 
	 * DESIGNER IMPACT:
	 * Normals critical for water appearance
	 * Incorrect normals = flat, unconvincing water
	 * Smooth normals = calm water feel
	 * Rough normals = turbulent, choppy feel
	 * 
	 * PERFORMANCE:
	 * Another GPU compute shader pass
	 * ~0.3ms for 512x512
	 * 
	 * TEXTURE FORMAT:
	 * Output: RGBA16F for normal map
	 * Why not RGBA8?
	 * - Water shader needs precision for lighting
	 * - 8-bit gives visible banding artifacts
	 * - 16-bit provides smooth gradients
	 * 
	 * TODO: Dispatch compute shader for normals
	 * Input: HeightMapTexture
	 * Output: NormalMapTexture
	 * 
	 * Algorithm:

	 * 2. Compute finite differences (slopes)
	 * 3. Cross product to get normal
	 * 4. Normalize and store
	 */

	 * SHADER PSEUDOCODE:

	 *     // Compute slopes
	 *     float slope_x = (h_right - h_center) / dx;

/**
 * ============================================================================
 * LAYER 2: SHALLOW WATER EQUATIONS - IMPLEMENTATION
 * ============================================================================
 * 
 * PHYSICS FOUNDATION:
 * Shallow Water Equations model fluid dynamics when depth << wavelength
 * Perfect for rivers, ponds, lakes, and ocean ripples
 * 
 * EQUATIONS:
 * dh/dt = -(d(hu)/dx + d(hv)/dy)          [Height continuity]
 * d(hu)/dt = -d(hu²/h)/dx - g·h·dh/dx     [X-momentum]
 * d(hv)/dt = -d(hv²/h)/dy - g·h·dh/dy     [Y-momentum]
 * 
 * VARIABLES:
 * h = water height above base
 * u, v = horizontal velocity components
 * g = gravity (9.81 m/s²)
 * 
 * PHYSICAL MEANING:
 * 1. Continuity: Water height changes based on flow divergence
 * 2. Momentum: Velocity changes from pressure gradients (height slopes)
 * 3. Together: Create wave propagation and damping
 * 
 * NUMERICAL METHOD:
 * Finite difference scheme on GPU
 * Each cell updates based on neighboring cells
 * 2nd order spatial accuracy, 1st order temporal
 * 
 * STABILITY REQUIREMENT (CFL Condition):
 * dt < dx / c, where c = sqrt(g*h)
 * If violated: Simulation becomes unstable (exponential error growth)
 * Function EnforceCFLStability ensures compliance
 * 
 * ============================================================================
 */

FShallowWaterSimulation::FShallowWaterSimulation(int32 InGridWidth, int32 InGridHeight, float InCellSize)
	: GridWidth(InGridWidth)
	, GridHeight(InGridHeight)
	, CellSize(InCellSize)
{

	 * Constructor - Store grid parameters
	 * 
	 * GRID SIZING CONSIDERATIONS:
	 *

	 *   128: Fast but coarse (~2ms)
	 *   256: Balanced (~3.5ms) - DEFAULT
	 *   512: Detailed but slow (~8ms)
	 * 
	 * @param InGridHeight Number of cells in Y direction
	 *   Usually matches GridWidth for square coverage
	 * 
	 * @param InCellSize Physical size of each cell (UE units)
	 *   100: Typical (1 cell = 1m at 100 UU/m scale)
	 *   50: Fine detail, 2x slower
	 *   200: Coarse, 2x faster

	 * DESIGNER GUIDANCE:
	 * - Start with 256x256, 100 cm cells
	 * - If performance poor, reduce to 128x128

void FShallowWaterSimulation::InitializeRHI(FRHICommandListBase& RHICmdList)
{
	/**
	 * Initialize GPU textures - RENDER THREAD
	 * 
	 * Creates double-buffered texture sets for stability
	 * 
	 * DOUBLE BUFFERING EXPLANATION:
	 * Problem: If compute shader reads and writes same texture:
	 * - Reads might get partially updated values
	 * - Creates race conditions
	 * - Simulation becomes incorrect/unstable
	 * 
	 * Solution: Use two textures per data type
	 * - Current: Read from this frame's data
	 * - Temp: Write new data here
	 * - Next frame: Swap (Temp becomes Current)
	 * 
	 * This guarantees:
	 * - All cells read consistent "old" data
	 * - All cells write to independent "new" buffer
	 * - No data races, deterministic results
	 * 
	 * TEXTURE ORGANIZATION:
	 * HeightField: Stores water surface height
	 *   R: Height value (cm)
	 *   G/B/A: Can store other data (foam, etc.)
	 * 
	 * VelocityField: Stores water velocity
	 *   R: Velocity X component (cm/s)
	 *   G: Velocity Y component (cm/s)
	 *   B/A: Reserved for future use
	 * 
	 * ALL TEXTURES:
	 * - RenderTargetable: Can render to
	 * - ShaderResource: Can sample from shaders
	 * - UAV: Can compute shader read/write
	 */

	// Height field textures
	// Current frame heights (read) and next frame heights (write)
	FRHITextureCreateDesc HeightDesc =
		FRHITextureCreateDesc::Create2D(TEXT("SWEHeightField"))
		.SetExtent(GridWidth, GridHeight)                            // Grid dimensions
		.SetFormat(PF_FloatRGBA)                                     // R32G32B32A32F precision
		.SetFlags(ETextureCreateFlags::RenderTargetable | 
				  ETextureCreateFlags::ShaderResource | 
				  ETextureCreateFlags::UAV);

	HeightFieldTexture = RHICreateTexture(HeightDesc);
	HeightFieldTemp = RHICreateTexture(HeightDesc);  // Double buffer

	// Velocity field textures
	// Current frame velocities (read) and next frame velocities (write)
	FRHITextureCreateDesc VelDesc =
		FRHITextureCreateDesc::Create2D(TEXT("SWEVelocityField"))
		.SetExtent(GridWidth, GridHeight)
		.SetFormat(PF_FloatRGBA)
		.SetFlags(ETextureCreateFlags::RenderTargetable | 
				  ETextureCreateFlags::ShaderResource | 
				  ETextureCreateFlags::UAV);

	VelocityFieldTexture = RHICreateTexture(VelDesc);
	VelocityFieldTemp = RHICreateTexture(VelDesc);  // Double buffer

	/**
	 * INITIALIZATION STATE:
	 * All textures start with zero values (cleared by GPU driver)
	 * This means:
	 * - All heights = 0 (flat calm water)
	 * - All velocities = 0 (no flow)
	 * - Water stays calm until impacts are applied
	 * 
	 * First frame:
	 * - Water remains flat
	 * - Ready to accept impacts (ApplyImpact calls)
	 * - Physics simulation begins next frame
	 * 
	 * MEMORY FOOTPRINT:
	 * At 256x256:
	 * - 4 textures × (256×256×16 bytes) = 4 MB
	 * - Acceptable GPU overhead
	 * 
	 * At 512x512:
	 * - 4 textures × (512×512×16 bytes) = 16 MB
	 * - Still acceptable but approaching limit
	 */
}

void FShallowWaterSimulation::UpdateSimulation(float DeltaTime)
{
	/**
	 * Update SWE physics one frame
	 * 
	 * SIMULATION PIPELINE:
	 * 
	 * 1. Enforce CFL Stability
	 *    Reduce DeltaTime if needed to prevent numerical instability
	 *    This is CRITICAL for stable simulation
	 *    Failure to do this = explosive growth of values
	 * 
	 * 2. Compute Height Continuity
	 *    dh/dt = -(d(hu)/dx + d(hv)/dy)
	 *    Updates height field based on velocity divergence
	 *    If water flows outward, height decreases
	 *    If water flows inward, height increases
	 * 
	 * 3. Compute Momentum Equations
	 *    d(hu)/dt = -d(hu²/h)/dx - g·h·dh/dx
	 *    d(hv)/dt = -d(hv²/h)/dy - g·h·dh/dy
	 *    Updates velocity based on pressure gradients
	 *    Height bumps create pressure that pushes water
	 * 
	 * PHYSICS INTUITION:
	 * - Heights want to flatten (pressure gradient)
	 * - Moving water wants to keep moving (convection)
	 * - Gravity pulls down (captured in pressure term)
	 * - These forces interact to create realistic waves
	 * 
	 * @param DeltaTime Requested time step (may be reduced for stability)
	 * 
	 * PERFORMANCE:
	 * - EnforceCFLStability: negligible (CPU calculation)
	 * - ComputeHeightContinuity: ~1-2 ms (GPU)
	 * - ComputeMomentumEquations: ~1-2 ms (GPU)
	 * - Total: ~3-5 ms for 256x256 grid
	 * 
	 * DESIGNER NOTES:
	 * If water looks weird/unstable:
	 * - Reduce grid cell size (more precision)
	 * - Increase grid resolution (more cells)
	 * - Reduce damping (makes it bouncier)
	 */

	// SAFETY FIRST: Ensure stability
	float SafeDeltaTime = DeltaTime;
	EnforceCFLStability(SafeDeltaTime);

	/**
	 * CFL EXPLANATION:
	 * Courant-Friedrichs-Lewy condition
	 * Ensures information doesn't propagate further than one grid cell per timestep
	 * If violated: Numerical errors grow exponentially each frame
	 * Result: Simulation explodes (values become huge)
	 * 
	 * Example violation:
	 * - Wave speed: 10 m/s = 1000 cm/s
	 * - Grid cell: 100 cm
	 * - Max dt: 100/1000 = 0.1 s
	 * - If frame time > 0.1s: Reduce to 0.05s (multiple sub-steps)
	 * 
	 * At 60 FPS (16.67ms):
	 * - SafeDeltaTime typically stays full 16.67ms
	 * - Only reduced if water very shallow or grid very fine
	 * - Usually not a problem in practice
	 */

	// GPU COMPUTE: Update height field
	ComputeHeightContinuity(SafeDeltaTime);

	// GPU COMPUTE: Update velocity field
	ComputeMomentumEquations(SafeDeltaTime);

	/**
	 * RESULT:
	 * After these calls:
	 * - HeightFieldTexture contains updated heights
	 * - VelocityFieldTexture contains updated velocities
	 * - Both ready for next frame and rendering
	 * - Values propagate outward from impact points as waves
	 */
}

void FShallowWaterSimulation::ComputeHeightContinuity(float DeltaTime)
{
	/**
	 * Compute Height Continuity Equation on GPU
	 * 
	 * EQUATION: dh/dt = -(d(hu)/dx + d(hv)/dy)
	 * 
	 * PHYSICAL MEANING:
	 * Water height changes based on flow divergence
	 * - If water flows outward: height decreases (water thins out)
	 * - If water converges inward: height increases (water piles up)
	 * - Divergence = how much flow spreads apart
	 * 
	 * NUMERICAL METHOD: Finite Difference Stencil
	 * 
	 * For each grid cell, compute:
	 * h_new = h_old - dt/dx * [flux_east - flux_west]
	 *                - dt/dy * [flux_north - flux_south]
	 * 
	 * Where flux = h * velocity
	 * 
	 * Visualized as stencil pattern:
	 * 
	 *            N
	 *            |
	 *        W - C - E
	 *            |
	 *            S
	 * 
	 * C = center cell (being updated)
	 * N/S/E/W = neighboring cells (read only)
	 * 
	 * IMPLEMENTATION ON GPU:
	 * - Launch 256x256 threads (one per cell)
	 * - Each thread:
	 *   1. Read neighboring heights and velocities
	 *   2. Compute divergence
	 *   3. Update height
	 *   4. Write to temporary texture
	 * - GPU parallelism: All 65k cells computed simultaneously
	 * 
	 * PERFORMANCE:
	 * - 256x256 grid: ~1-2 ms (GPU very fast at this)
	 * - 512x512 grid: ~3-4 ms (4x more cells)
	 * - Limited by GPU memory bandwidth, not computation
	 * 
	 * STABILITY:
	 * Must use small enough dt (CFL condition)
	 * Otherwise: Numerical diffusion creates instability
	 * EnforceCFLStability ensures this
	 * 
	 * @param DeltaTime Time step (guaranteed safe by caller)
	 * 
	 * TODO: Dispatch compute shader
	 * Input: VelocityFieldTexture (read velocities)
	 * Input/Output: HeightFieldTexture (read old, write new)
	 * Uses: CellSize, DeltaTime
	 * 
	 * SHADER PSEUDOCODE:
	 * 
	 * [numthreads(16, 16, 1)]
	 * void HeightContinuityShader(uint3 id : SV_DispatchThreadID)
	 * {
	 *     int x = id.x;
	 *     int y = id.y;
	 *     
	 *     // Read velocities from neighbors
	 *     float flux_east = Velocity[x+1][y].x * Height[x+1][y];
	 *     float flux_west = Velocity[x-1][y].x * Height[x-1][y];
	 *     float flux_north = Velocity[x][y+1].y * Height[x][y+1];
	 *     float flux_south = Velocity[x][y-1].y * Height[x][y-1];
	 *     
	 *     // Compute divergence
	 *     float divergence = (flux_east - flux_west) / CellSize +
	 *                        (flux_north - flux_south) / CellSize;
	 *     
	 *     // Update height
	 *     float h_new = Height[x][y] - DeltaTime * divergence;
	 *     
	 *     // Apply damping (energy loss)
	 *     h_new *= Damping;
	 *     
	 *     // Write result
	 *     HeightFieldTemp[x][y] = h_new;
	 * }
	 * 
	 * DAMPING EXPLANATION:
	 * Multiplying by Damping (0.99) each frame:
	 * - After 1 frame: 100% * 0.99 = 99% (1% lost)
	 * - After 60 frames: 100% * 0.99^60 ≈ 55% (half energy)
	 * - After 150 frames: 100% * 0.99^150 ≈ 22% (quarter)
	 * 
	 * This creates realistic wave decay
	 * No damping = waves persist forever (unrealistic)
	 * High damping = quick fade (feels dead)
	 * Tuned value = natural decay (feels right)

void FShallowWaterSimulation::ComputeMomentumEquations(float DeltaTime)
{
	/**
	 * Compute Momentum Equations on GPU
	 * 
	 * EQUATIONS:
	 * d(hu)/dt = -d(hu²/h)/dx - g·h·dh/dx
	 * d(hv)/dt = -d(hv²/h)/dy - g·h·dh/dy
	 * 
	 * PHYSICAL MEANING (for X-component):
	 * - First term: Convective acceleration (water pushes itself)
	 * - Second term: Pressure gradient (slopes in height push water)
	 * - Together: Water accelerates/decelerates based on flow and pressure
	 * 
	 * TWO FORCE MECHANISMS:
	 * 
	 * 1. CONVECTIVE ACCELERATION
	 *    d(u²)/dx = how steeply velocity is increasing
	 *    If water flows fast into slower water: acceleration spike
	 *    Creates shock-like behavior, breaks waves realistically
	 * 
	 * 2. PRESSURE GRADIENT
	 *    g·dh/dx = gravity × slope
	 *    Height bumps (high spots) push water away (downhill)
	 *    This is the restoring force that creates oscillation (waves!)
	 *    Larger slope = stronger force = water moves faster
	 * 
	 * HOW THEY WORK TOGETHER:
	 * - Pressure gradient tries to flatten bumps
	 * - Convection can steepen them (creates complex dynamics)
	 * - Result: Realistic wave propagation and breaking
	 * 
	 * NUMERICAL METHOD:
	 * Similar finite difference stencil as height equation
	 * Update velocity based on pressure gradients
	 * 
	 * GPU IMPLEMENTATION:
	 * Another 256x256 thread grid
	 * Each thread:
	 * 1. Read height and velocity from 3x3 neighborhood
	 * 2. Compute pressure gradient term
	 * 3. Compute convective term
	 * 4. Update velocity
	 * 5. Write to temporary velocity texture
	 * 
	 * PERFORMANCE: ~1-2 ms (similar to height computation)
	 * 
	 * @param DeltaTime Time step (guaranteed safe)
	 * 
	 * TODO: Dispatch compute shader
	 * Input: HeightFieldTexture, VelocityFieldTexture
	 * Output: VelocityFieldTemp
	 * Uses: CellSize, Gravity, DeltaTime
	 * 
	 * SHADER PSEUDOCODE:
	 * 
	 * [numthreads(16, 16, 1)]
	 * void MomentumShader(uint3 id : SV_DispatchThreadID)
	 * {
	 *     int x = id.x;
	 *     int y = id.y;
	 *     
	 *     // Read center and neighbors
	 *     float h = Height[x][y];
	 *     float h_east = Height[x+1][y];
	 *     float u = Velocity[x][y].x;
	 *     
	 *     // Pressure gradient term: -g * h * dh/dx
	 *     float pressure_grad = -Gravity * h * (h_east - h) / CellSize;
	 *     
	 *     // Convective term: -d(u²)/dx
	 *     float u_east = Velocity[x+1][y].x;
	 *     float convection = -(u_east*u_east - u*u) / (2*CellSize);
	 *     
	 *     // Update velocity
	 *     float u_new = u + DeltaTime * (pressure_grad + convection) / h;

	 *     // Cap velocity (prevent overflow)
	 *     u_new = clamp(u_new, -MaxVelocity, MaxVelocity);

	 *     // Write result

	 *

	 * Convective term can cause instability if not careful
	 * This is why CFL condition is strict
	 * Too large dt = error compounds
	 * Small dt = stable but slower computation

void FShallowWaterSimulation::EnforceCFLStability(float& SafeDeltaTime)
{
	/**
	 * Enforce Courant-Friedrichs-Lewy Stability Condition
	 * 
	 * WHAT IS CFL?
	 * Mathematical requirement for stable finite difference schemes
	 * Ensures numerical errors don't grow exponentially
	 * 
	 * PRINCIPLE:
	 * Information propagates at wave speed c
	 * In one time step, should not travel more than one grid cell
	 * 
	 * FORMULA: dt < dx / c
	 * 
	 * For water:
	 * c = sqrt(g * h)  (shallow water wave speed)
	 * 
	 * EXAMPLE CALCULATION:
	 * - Gravity: g = 9.81 m/s² = 981 cm/s²
	 * - Water depth: h = 100 cm (1 meter)
	 * - Wave speed: c = sqrt(981 * 100) ≈ 313 cm/s ≈ 3.1 m/s
	 * - Grid cell: dx = 100 cm
	 * - Max dt: 100 / 313 ≈ 0.32 seconds
	 * - With 0.5 safety factor: dt < 0.16 seconds
	 * 
	 * At 60 FPS: dt = 16.67 ms = 0.01667 seconds
	 * This is well below 0.16s limit (safe!)
	 * 
	 * WHEN IS CFL VIOLATED?
	 * - Very shallow water (h small, c small, dt max small)
	 * - Very fine grid (dx small, dt max small)
	 * - Very high frame rate (dt large, approaches limit)
	 * 
	 * WHAT HAPPENS IF VIOLATED?
	 * - Numerical errors grow each frame
	 * - Wave speeds become incorrect (distorted)
	 * - Values become unreasonably large (explosion)
	 * - Simulation crashes
	 * 
	 * THIS FUNCTION:
	 * Prevents violation by reducing SafeDeltaTime if needed
	 * Conservative approach: Always safe, never unstable
	 * 
	 * @param SafeDeltaTime [in/out]
	 *   Input: Requested time step from frame
	 *   Output: Reduced time step if CFL violated
	 *           Same value if already safe
	 * 
	 * DESIGNER IMPACT:
	 * If you see "water is unstable/exploding":
	 * 1. Reduce grid cell size (CellSize parameter)
	 * 2. Reduce grid resolution (use 128x128 instead 256x256)
	 * 3. Reduce damping (less energy loss = more stable)
	 * 
	 * PROGRAMMER OPTIMIZATION:
	 * This function computes conservative max_dt every frame
	 * Could cache if parameters don't change
	 * But recomputing every frame is safer (handles shallow/deep spots)
	 */

	const float MaxWaveSpeed = 20.0f;  // Typical max wave speed (cm/s)
	const float SafetyFactor = 0.5f;   // Extra safety margin

	/**
	 * WAVE SPEED EXPLANATION:
	 * For shallow water: c = sqrt(g*h)
	 * 
	 * Typical scenarios:
	 * - Deep water (h=1000cm): c = sqrt(981*1000) ≈ 313 cm/s ≈ 3.1 m/s
	 * - Mid-depth (h=100cm): c = sqrt(981*100) ≈ 99 cm/s ≈ 1 m/s
	 * - Shallow (h=10cm): c = sqrt(981*10) ≈ 31 cm/s
	 * - Very shallow (h=1cm): c = sqrt(981*1) ≈ 10 cm/s
	 * 
	 * Worst case (maximum wave speed):
	 * Assuming depth up to ~4 meters (400cm):
	 * c_max = sqrt(981 * 400) ≈ 200 cm/s = 20 m/s (but wave speed would be reduced by damping)
	 * 
	 * So MaxWaveSpeed = 20 cm/s is conservative estimate
	 * (actual wave speeds usually lower due to damping)
	 * 
	 * SAFETY FACTOR:
	 * CFL says dt < dx / c
	 * We multiply by 0.5 factor to be extra safe
	 * This is standard practice in CFD codes
	 * Ensures stability even with estimation errors
	 */

	float MaxDeltaTime = (CellSize / MaxWaveSpeed) * SafetyFactor;

	// Reduce if frame time exceeds safe limit
	SafeDeltaTime = FMath::Min(SafeDeltaTime, MaxDeltaTime);

	/**
	 * WHAT THIS DOES:
	 * If SafeDeltaTime > MaxDeltaTime:
	 *   Reduce to MaxDeltaTime (prevent explosion)
	 * Otherwise:
	 *   Keep unchanged (frame time is safe)
	 * 
	 * EXAMPLE:
	 * MaxDeltaTime = 0.05 seconds (50ms)
	 * Frame time = 16.67ms
	 * Result: SafeDeltaTime = 16.67ms (fine, unchanged)
	 * 
	 * Another example (if someone made frame rate 10 FPS):
	 * Frame time = 100ms
	 * Result: SafeDeltaTime = 50ms (reduced by this function)
	 * Simulation runs 2 sub-steps to stay stable
	 * 
	 * DESIGNER NOTE:
	 * If water looks weird/unstable:
	 * 1. This function is working (preventing explosion)
	 * 2. Solution: Use larger grid cells or smaller grid
	 * 3. Or: Accept slower update rate (30 Hz instead 60 Hz)
	 */
}

/**
 * ============================================================================
 * LAYER 3: SPH PARTICLES - IMPLEMENTATION
 * ============================================================================
 * 
 * PHYSICS FOUNDATION:
 * SPH (Smoothed Particle Hydrodynamics) represents fluid as particles
 * Each particle has mass, density, pressure, velocity
 * Forces propagate between particles via kernel functions
 * 
 * ADVANTAGES OF SPH:
 * - Easy to implement (no complex grids)
 * - Natural particle effects (splashes look great)
 * - Handles free surfaces naturally
 * - Can create dramatic visual interactions
 * 
 * DISADVANTAGES:
 * - Slower than grid-based methods (O(n²) vs O(n))
 * - Needs many particles for smooth results
 * - Limited to small areas (can't simulate entire ocean)
 * 
 * SOLUTION: Use SPH for LOCAL INTERACTIONS ONLY
 * - Combine with SWE for physics
 * - Combine with FFT for visuals
 * - SPH only when camera is very close
 * 
 * COMPLEXITY:
 * - Neighbor search: O(n²) but n ≤ 256
 * - Density computation: O(n * k) where k = avg neighbors
 * - Force computation: Similar
 * - Integration: O(n)
 * 
 * TOTAL: ~256² / 2 = 32k operations = ~1.5ms CPU
 * Acceptable performance for high-quality splashes
 * 
 * ============================================================================
 */

FLocalSPHInteraction::FLocalSPHInteraction(int32 InMaxParticles)
	: MaxParticles(InMaxParticles)
{
	/**
	 * Constructor - Setup particle storage
	 * 
	 * @param InMaxParticles Maximum particle count (default 256)
	 *   32: Sparse splashes (fast, low detail)
	 *   128: Medium splashes (good balance)
	 *   256: Dense splashes (default, smooth)
	 *   512: Very dense (slow, maximum quality)
	 * 
	 * MEMORY FOOTPRINT:
	 * Per particle: ~64 bytes (position, velocity, density, pressure)
	 * 256 particles: 16 KB (tiny)
	 * Can hold thousands without memory issues
	 * 
	 * PERFORMANCE IS THE LIMITING FACTOR:
	 * Not memory, but computation time
	 * More particles = O(n²) neighbor search
	 * 256 particles = acceptable 1.5ms
	 * 512 particles = pushing it at 3ms
	 * 1024 particles = too slow for 60 FPS
	 * 
	 * DESIGNER GUIDANCE:
	 * Start with 256, reduce to 128 if performance issues
	 * Increase to 512 only if you have FPS budget
	 * Most players won't notice detail difference
	 * 
	 * Reserve memory upfront for efficiency:
	 * Avoid dynamic reallocation during simulation
	 * Pre-allocate at max capacity
	 */

	Particles.Reserve(MaxParticles);
}

FLocalSPHInteraction::~FLocalSPHInteraction()

	/**

	 * No GPU resources to release
	 * All CPU-based in this implementation

}

}

void FLocalSPHInteraction::UpdateParticles(float DeltaTime, const FVector& WaterPlaneCenter)
{
	/**
	 * Update all particles one frame
	 * 
	 * SPH PIPELINE:
	 * 1. ComputeDensity: For each particle, sum neighbors
	 * 2. ComputePressure: Calculate pressure from density
	 * 3. ComputeForces: Calculate forces on each particle
	 * 4. Integrate: Update positions and velocities
	 * 
	 * ITERATIONS:
	 * Some SPH implementations use multiple sub-iterations
	 * This version: One pass per frame
	 * Trade-off: Faster but less accurate
	 * 
	 * For splash effects: One pass is sufficient
	 * Visual quality still looks great
	 * 
	 * @param DeltaTime Frame time (for integration step)
	 * @param WaterPlaneCenter Center of water volume (for bounds checking)
	 *   Used to remove particles that leave water volume
	 *   Prevents unbounded particle growth
	 * 
	 * PERFORMANCE:
	 * Early return if no particles (nothing to do)
	 * SavesCPU cycles on frames with no splashes
	 * 
	 * Total time for 256 particles:
	 * - FindNeighbors: O(n²) = ~65k comparisons = 1ms
	 * - ComputeDensity: ~0.2ms
	 * - ComputePressure: ~0.1ms
	 * - ComputeForces: ~0.5ms (simplified)
	 * - Integrate: ~0.2ms
	 * - Total: ~2ms (pessimistic estimate)
	 * Actual: Usually ~1-1.5ms
	 */

	if (Particles.Num() == 0) return;  // Nothing to simulate

	// DENSITY: Kernel summation
	ComputeDensity();

	// PRESSURE: Equation of state
	ComputePressure();

	// FORCES: Pressure gradients + viscosity
	ComputeForces();

	// INTEGRATION: Update positions/velocities
	Integrate(DeltaTime);
}

void FLocalSPHInteraction::ApplyImpact(const FVector& ImpactLocation, float ImpactForce)
{
	/**
	 * Create splash at impact location
	 * 
	 * USAGE:
	 * - Character jumps in water
	 * - Projectile hits water surface
	 * - Object falls into water
	 * 
	 * HOW IT WORKS:
	 * 1. Particle count scales with impact force
	 *    Stronger impact = more particles
	 *    Creates scaled visual effect
	 * 
	 * 2. Initial velocity upward (splash shoots up)
	 *    Related to impact force
	 *    Stronger impact = faster particles
	 * 
	 * 3. Call AddParticleSource to emit particles
	 *    Handles actual particle creation
	 * 
	 * @param ImpactLocation Where splash occurs
	 * @param ImpactForce How hard (affects particle count and speed)
	 * 
	 * FORCE SCALING:
	 * ParticleCount = ImpactForce * 0.1
	 * InitialVelocity = ImpactForce * 0.5
	 * 
	 * Example:
	 * ImpactForce = 500 (character jump)
	 *   → 50 particles
	 *   → Initial velocity 250 cm/s = 2.5 m/s up
	 *   → Nice splashy effect
	 * 
	 * ImpactForce = 2000 (explosion)
	 *   → 200 particles (capped at MaxParticles=256)
	 *   → Initial velocity 1000 cm/s = 10 m/s up
	 *   → Massive spray effect
	 * 
	 * DESIGNER NOTES:
	 * Impact force should scale with event magnitude:
	 * - Gentle ripple: Force = 10-50
	 * - Character jump: Force = 300-1000
	 * - Explosion: Force = 2000-10000
	 * - Tidal wave: Force = 20000+
	 * 
	 * Experiment to find values that look good
	 * Different water and physics scales need different ranges
	 */

	// Particle count scales with force
	// Clamp between 1 (always spawn something) and 32 (max per impact)
	int32 ParticleCount = FMath::Clamp(FMath::RoundToInt(ImpactForce * 0.1f), 1, 32);

	// Initial velocity upward (away from impact)
	// Scaled with force magnitude
	FVector InitialVelocity = FVector::UpVector * ImpactForce * 0.5f;

	// Create particles
	AddParticleSource(ImpactLocation, InitialVelocity, ParticleCount);
}

void FLocalSPHInteraction::AddParticleSource(const FVector& SourceLocation, const FVector& SourceVelocity, int32 ParticleCount)
{
	/**
	 * Emit particles from source location
	 * 
	 * Creates ParticleCount new particles around SourceLocation
	 * Each with initial velocity SourceVelocity + randomness
	 * 
	 * WHY RANDOMNESS?
	 * Without it: Particles in straight line (unnatural)
	 * With it: Natural spray pattern (water-like)
	 * 
	 * PARTICLE SPAWNING PATTERN:
	 * Center: SourceLocation
	 * Spread: ±25 cm in X/Y, 0-50 cm in Z
	 * Result: Cone-shaped spray
	 * 
	 * @param SourceLocation Where particles spawn
	 * @param SourceVelocity Bulk velocity direction
	 * @param ParticleCount How many to create
	 * 
	 * LOOP TERMINATION:
	 * Two conditions for when to stop:
	 * 1. Created ParticleCount particles
	 * 2. Reached MaxParticles limit
	 * 
	 * Prevents array from growing unbounded
	 * Ensures O(n²) stays tractable
	 * 
	 * PERFORMANCE:
	 * Loop: O(ParticleCount) = O(32) = fast
	 * Array allocation: May trigger growth if needed
	 * Typically no reallocation (preallocated to MaxParticles)
	 * 
	 * DESIGNER IMPACT:
	 * More particles = more detailed splash
	 * But also = more impact force needed
	 * Tuning forces to desired effect is designer's job
	 */

	for (int32 i = 0; i < ParticleCount && Particles.Num() < MaxParticles; ++i)
	{
		// Create new particle and get reference
		FSPHParticle& NewParticle = Particles.AddDefaulted_GetRef();

		/**
		 * POSITION INITIALIZATION:
		 * Spawn around source point with random offset
		 * Creates natural spread pattern
		 */

		NewParticle.Position = SourceLocation + FVector(
			FMath::FRand() * 50.0f - 25.0f,      // ±25 cm in X (random)
			FMath::FRand() * 50.0f - 25.0f,      // ±25 cm in Y (random)
			FMath::FRand() * 50.0f               // 0-50 cm in Z (upward bias)
		);

		/**
		 * VELOCITY INITIALIZATION:
		 * Source velocity + random jitter
		 * Creates chaotic splash appearance
		 */

		NewParticle.Velocity = SourceVelocity + FVector(
			FMath::FRandRange(-50.0f, 50.0f),     // ±50 cm/s random X
			FMath::FRandRange(-50.0f, 50.0f),     // ±50 cm/s random Y
			FMath::FRandRange(0.0f, 100.0f)       // Extra upward randomness
		);

		/**
		 * PROPERTY INITIALIZATION:
		 * Density/Pressure will be computed next frame
		 */

		NewParticle.Density = RestDensity;
		NewParticle.Pressure = 0.0f;

		/**
		 * NEXT FRAME:
		 * ComputeDensity will find neighbors
		 * ComputePressure will create pressure field
		 * ComputeForces will push particles apart
		 * Integrate will move particles
		 * Result: Realistic fluid splash
		 */
	}
}

FVector FLocalSPHInteraction::GetInteractionForceAt(const FVector& QueryLocation, float QueryRadius) const
{
	/**
	 * Query net interaction force at location
	 * 
	 * Used for physics integration (pushing objects)
	 * 
	 * USAGE:
	 * Called from ApplyBuoyancyToActor
	 * Actor gets pushed by nearby splash particles
	 * Creates interaction between water particles and actors
	 * 
	 * ALGORITHM:
	 * For each particle in range:
	 *   1. Calculate distance to query point
	 *   2. Apply falloff (force decreases with distance)
	 *   3. Calculate force magnitude from pressure
	 *   4. Add to total force
	 * 
	 * FORCE DIRECTION:
	 * Away from particle (repulsive)
	 * Particles pushed back repel objects forward
	 * Creates realistic collision-like behavior
	 * 
	 * FALLOFF:
	 * Linear falloff: Force = 1.0 at center, 0.0 at edge
	 * Could use other falloffs (exponential, etc.)
	 * Linear keeps computation simple
	 * 
	 * @param QueryLocation Where to query force
	 * @param QueryRadius How far to search (cm)
	 * @return Net force vector (in Newtons)
	 * 
	 * PERFORMANCE:
	 * Loop: O(n) = O(256) = fast
	 * Most particles will be outside radius (early continue)
	 * Typical: 10-50 particles in range
	 * 
	 * DESIGNER USAGE:
	 * QueryRadius should be ~500 cm (smooth interaction)
	 * Smaller radius = sharper interaction
	 * Larger radius = more influence from splashes
	 * 
	 * PHYSICS NOTES:
	 * Force is pressure-based (not momentum-based)
	 * More realistic than simple velocity transfer
	 * Particles with high pressure push harder
	 */

	FVector TotalForce = FVector::ZeroVector;

	for (const FSPHParticle& Particle : Particles)
	{
		// Distance from particle to query location
		float Distance = FVector::Dist(QueryLocation, Particle.Position);

		// Skip if too far or too close (avoid singularities)
		if (Distance < QueryRadius && Distance > KINDA_SMALL_NUMBER)
		{
			/**
			 * FORCE CALCULATION:
			 * F = Pressure / Density * Falloff
			 * 
			 * - Pressure/Density: Specific force (force per unit mass)
			 * - Falloff: Linear distance attenuation
			 * 
			 * Why Pressure/Density?
			 * More fundamental than raw pressure
			 * Accounts for particle density variations
			 * Creates correct force scaling
			 */

			float ForceMagnitude = (Particle.Pressure / (Particle.Density + KINDA_SMALL_NUMBER)) * 
									(1.0f - Distance / QueryRadius);  // Linear falloff

			// Force direction: away from particle (repulsive)
			FVector Direction = (Particle.Position - QueryLocation).GetSafeNormal();

			// Add to total
			TotalForce += Direction * ForceMagnitude;
		}
	}

	return TotalForce;
}

void FLocalSPHInteraction::FindNeighbors(int32 ParticleIndex, TArray<int32>& OutNeighbors)
{
	/**
	 * Find all particles within SmoothingRadius
	 * 
	 * USED FOR:
	 * Kernel summation in density/pressure/force computation
	 * Only nearby particles contribute to forces
	 * This is what makes SPH local (scalable)
	 * 
	 * ALGORITHM: Brute force O(n) per particle
	 * For 256 particles: 256² = 65k comparisons
	 * But most return early (too far)
	 * Average k neighbors << n
	 * Total: O(n*k) where k << n
	 * 
	 * OPTIMIZATION OPPORTUNITIES:
	 * - Spatial hash: O(n) instead of O(n²)
	 * - For 256 particles: Not worth overhead (brute force faster)
	 * - For 10k particles: Spatial hash becomes necessary
	 * 
	 * CURRENT APPROACH:
	 * Simple, predictable, no cache misses
	 * Perfect for small particle counts
	 * 
	 * @param ParticleIndex Which particle to find neighbors for
	 * @param OutNeighbors [out] Indices of neighbors
	 * 
	 * PERFORMANCE:
	 * - Loop: O(n) = O(256) = ~1 microsecond per particle
	 * - Distance check: ~10 ns
	 * - Called 256 times = ~256 microseconds total
	 * - Very fast compared to overall simulation
	 * 
	 * DESIGNER NOTE:
	 * SmoothingRadius controls splash extent
	 * Larger radius = bigger splash
	 * 100 cm typical (1 meter interaction range)
	 * 
	 * PROGRAMMER NOTE:
	 * If profiling shows bottleneck:
	 * 1. Add spatial hash grid
	 * 2. Or: Limit to closer neighbors only
	 * 3. Or: Compute less frequently (every Nth frame)
	 */

	OutNeighbors.Empty();
	const FSPHParticle& Particle = Particles[ParticleIndex];

	for (int32 i = 0; i < Particles.Num(); ++i)
	{
		if (i == ParticleIndex) continue;  // Skip self

		float Distance = FVector::Dist(Particle.Position, Particles[i].Position);
		if (Distance < SmoothingRadius)
		{
			OutNeighbors.Add(i);
		}
	}
}

void FLocalSPHInteraction::ComputeDensity()
{
	/**
	 * Compute particle density from neighbors
	 * 
	 * SPH FORMULA: ρ_i = Σ_j m_j W(r_ij)
	 * 
	 * WHERE:
	 * ρ_i = density of particle i
	 * m_j = mass of neighbor j
	 * W = kernel function (smoothing kernel)
	 * r_ij = distance between i and j
	 * 
	 * PHYSICAL MEANING:
	 * Density is weighted sum of nearby mass
	 * Closer neighbors count more (kernel falloff)
	 * Further neighbors count less
	 * Produces smooth density field
	 * 
	 * GAUSSIAN KERNEL:
	 * W(r) = exp(-(r/h)²)
	 * h = smoothing radius (SmoothingRadius = 100 cm)
	 * 
	 * - At r=0: W=1 (maximum)
	 * - At r=h: W=exp(-1) ≈ 0.37 (37%)
	 * - At r=2h: W=exp(-4) ≈ 0.02 (2%)
	 * 
	 * This creates smooth falloff
	 * No discontinuity at boundary
	 * Smooth forces result
	 * 
	 * COMPUTATION:
	 * For each particle:
	 *   1. Find neighbors
	 *   2. For each neighbor: compute kernel value
	 *   3. Multiply by neighbor mass
	 *   4. Sum all contributions
	 * 
	 * Result: Accurate density field
	 * 
	 * CLAMPING:
	 * Clamp to RestDensity minimum (prevents division by zero)
	 * RestDensity = 1000 kg/m³ (water)
	 * 
	 * Some scenarios can have zero density:
	 * - Isolated particle (no neighbors)
	 * - Particle at boundary
	 * - First frame before simulation
	 * 
	 * Clamping prevents undefined behavior in pressure computation
	 * 
	 * PERFORMANCE:
	 * For each particle: FindNeighbors + kernel summation
	 * O(n*k) where k = average neighbors
	 * Typical k = 20-40 neighbors per particle
	 * Total: ~0.2ms for 256 particles
	 * 
	 * DESIGNER IMPACT:
	 * Density determines fluid behavior
	 * Too uniform = can't detect interactions
	 * Too variable = instability
	 * Gaussian kernel provides good balance
	 */

	for (int32 i = 0; i < Particles.Num(); ++i)
	{
		// Find all neighbors of this particle
		TArray<int32> Neighbors;
		FindNeighbors(i, Neighbors);

		// Sum contributions from neighbors
		float Density = 0.0f;
		for (int32 NeighborIdx : Neighbors)
		{
			float Distance = FVector::Dist(Particles[i].Position, Particles[NeighborIdx].Position);

			/**
			 * GAUSSIAN KERNEL:
			 * W(r) = exp(-(r/h)²)
			 * 
			 * More complex kernels possible:
			 * - Cubic spline (smoother derivatives)
			 * - Poly6 (faster computation)
			 * - Wendland (more compact support)
			 * 
			 * Gaussian chosen for:
			 * - Simple computation
			 * - Smooth gradients (good forces)
			 * - No discontinuities (stable)
			 */

			float Kernel = FMath::Exp(-(Distance * Distance) / (SmoothingRadius * SmoothingRadius));

			// Accumulate density contribution
			Density += Particles[NeighborIdx].Mass * Kernel;
		}

		// Clamp to prevent division by zero
		Particles[i].Density = FMath::Max(Density, RestDensity);
	}
}

void FLocalSPHInteraction::ComputePressure()
{
	/**
	 * Compute particle pressure from density
	 * 
	 * EQUATION OF STATE: P = k * ((ρ/ρ₀)^γ - 1)
	 * 
	 * WHERE:
	 * P = pressure
	 * ρ = current density
	 * ρ₀ = rest density (target density, 1000 kg/m³ for water)
	 * k = gas stiffness constant (tuned)
	 * γ = polytropic exponent (7 for water)
	 * 
	 * PHYSICAL MEANING:
	 * - If ρ = ρ₀: Pressure = 0 (relaxed state)
	 * - If ρ > ρ₀: Pressure > 0 (overcrowded, pushes apart)
	 * - If ρ < ρ₀: Pressure < 0 (undercrowded, attracts)
	 * 
	 * This creates restoring forces
	 * Particles try to maintain rest density
	 * Prevents them from clustering or spreading
	 * 
	 * STIFFNESS PARAMETER k:
	 * Controls how strongly pressure responds to density variation
	 * 
	 * k = 20: Water is soft (compressible, waves easily)
	 * k = 50: Water is stiff (less compressible, stiffer response)
	 * 
	 * Higher k = stiffer, more resistant
	 * Game feel consideration (designer's choice)
	 * 
	 * POLYTROPIC EXPONENT γ = 7:
	 * Standard for weakly-compressible fluids
	 * Comes from gas dynamics theory
	 * γ = 1: Isothermal gas
	 * γ = 7: Typical for water SPH
	 * 
	 * WEAKLY-COMPRESSIBLE SPH:
	 * Allows small density variations (5-10%)
	 * Much more stable than incompressible SPH
	 * Avoids pressure Poisson solver (complex)
	 * Trade-off: Slight compressibility vs stability
	 * Players don't notice this compromise
	 * 
	 * PRESSURE RANGE:
	 * At rest: P = 0
	 * When crowded: P = 20 * ((1.1)^7 - 1) ≈ 23 (positive)
	 * When sparse: P = 20 * ((0.9)^7 - 1) ≈ -3 (negative)
	 * 
	 * Positive pressures push particles apart
	 * Negative pressures keep particles together
	 * Forces balance to maintain density
	 * 
	 * PERFORMANCE:
	 * O(n) operation (one pow per particle)
	 * Pow is relatively expensive (~100 CPU cycles)
	 * Total: ~0.1ms for 256 particles
	 * 
	 * OPTIMIZATION:
	 * Could use lookup table instead of pow()
	 * Trade: Memory (256 entries) for speed
	 * Not necessary for 256 particles (already fast)
	 * 
	 * DESIGNER NOTES:
	 * Pressure directly affects:
	 * - How tightly particles stay together
	 * - How quickly they push objects
	 * - Overall "stiffness" of water
	 * 
	 * Tuning k creates different water feel:
	 * k=10: Gooey, sticky water
	 * k=20: Normal water
	 * k=50: Stiff, bouncy water
	 * k=100: Very stiff, explosive water
	 */

	const float GasConstant = 20.0f;  // Stiffness (tuned for water)
	const float Gamma = 7.0f;         // Polytropic exponent

	for (FSPHParticle& Particle : Particles)
	{
		// Normalize density (ratio to rest density)
		float DensityRatio = Particle.Density / RestDensity;

		// Equation of state
		// (ρ/ρ₀)^γ computes stiff response to density change
		// -1 makes pressure zero at rest density
		Particle.Pressure = GasConstant * (FMath::Pow(DensityRatio, Gamma) - 1.0f);

		/**
		 * RESULT:
		 * Pressure field now represents how crowded each particle is
		 * Next step (ComputeForces) uses this to push particles
		 * Particles push away from high-pressure areas
		 * Result: Fluid particles scatter naturally
		 */
	}
}

void FLocalSPHInteraction::ComputeForces()
{
	/**
	 * Compute pressure and viscosity forces on each particle
	 * 
	 * FULL SPH FORMULA:
	 * F_i = -Σ_j m_j (P_i/ρ_i² + P_j/ρ_j²) ∇W_ij + 
	 *       + Σ_j m_j μ (v_j - v_i)/ρ_j ∇²W_ij
	 * 
	 * Two components:
	 * 1. PRESSURE FORCE (first term)
	 *    -m * (P/ρ²) * gradient(kernel)
	 *    Pushes particles apart when crowded
	 *    Gradient points away from neighbors
	 *    Creates spreading/scattering behavior
	 * 
	 * 2. VISCOSITY FORCE (second term)
	 *    m * μ * (v_neighbor - v_self) * laplacian(kernel)
	 *    Smooths velocity differences
	 *    Makes particles stick together slightly
	 *    Prevents shear instability
	 * 
	 * SIMPLIFIED IMPLEMENTATION:
	 * Full SPH requires computing kernel gradients/laplacians
	 * Complex and CPU-intensive
	 * 
	 * For splash effects at camera-close range:
	 * - Pressure forces are most important
	 * - Viscosity can be simplified or omitted
	 * - Visual quality still excellent
	 * 
	 * THIS VERSION: SIMPLIFIED (placeholder)
	 * Pressure forces should be computed
	 * Viscosity forces can be omitted or approximated
	 * 
	 * GRAVITY:
	 * Always downward (applied during Integrate)
	 * Not part of pressure/viscosity computation
	 * 
	 * PERFORMANCE TRADE-OFF:
	 * Full SPH: Very accurate, expensive
	 * Simplified: Good enough for splashes, fast
	 * 
	 * For game purposes: Simplified sufficient
	 * Players don't notice missing viscosity term

	 *   2. Store in particle.force for Integrate

	 * TODO: Implement viscosity force (if desired)
	 * Similar but uses velocity difference
	 * Laplacian of kernel instead of gradient

void FLocalSPHInteraction::Integrate(float DeltaTime)
{
	/**
	 * Update particle positions and velocities
	 * 
	 * INTEGRATION METHOD: Euler (simple, sufficient)
	 * 
	 * FORMULA:
	 * v_new = v_old + (F/m) * dt
	 * x_new = x_old + v_new * dt
	 * 
	 * APPLIED FORCES:
	 * 1. GRAVITY (always present)
	 *    F = m * g = 1g mass × 9.81 m/s² = 9.81 N (downward)
	 *    Simplified: just reduce velocity Z component
	 *    v.z -= 9.81 * dt
	 * 
	 * 2. PRESSURE FORCES (from ComputeForces)
	 *    Applied via accumulated force term
	 *    TODO: Uncomment when ComputeForces implemented
	 * 
	 * 3. VISCOSITY FORCES (optional)
	 *    Smooths motion, but not critical for splashes
	 * 
	 * GRAVITY EXPLANATION:
	 * 9.81 m/s² = 981 cm/s² (Unreal uses cm)
	 * After 1 frame (0.01667s): v.z -= 16.4 cm/s ≈ 0.164 m/s
	 * After 1 second (60 frames): particles falling ~5 m
	 * Realistic terminal velocity created
	 * 
	 * BOUNDARY HANDLING:
	 * Particles falling below Z = -1000 are removed
	 * Prevents unbounded particle array growth
	 * Also represents water "disappearing" (drainage/evaporation)
	 * 
	 * OPTIMIZATION OPPORTUNITY:
	 * Higher-order integration (RK2, RK4) more accurate
	 * But Euler sufficient for splash effects
	 * RK methods ~3x more expensive (not worth it)
	 * 
	 * @param DeltaTime Frame time in seconds
	 * 
	 * PERFORMANCE:
	 * O(n) = O(256) = very fast
	 * Dominated by earlier density/pressure computation
	 * Integration itself negligible
	 * 
	 * DESIGNER NOTES:
	 * Particles fall due to gravity
	 * With damping: Fall velocity limited
	 * Without damping: Particles accelerate forever (unrealistic)
	 * 
	 * Adjustment points:
	 * - Z boundary (-1000): When particles "disappear"
	 * - Gravity: How fast particles fall
	 * - Initial velocity: How high splashes go
	 */

	for (FSPHParticle& Particle : Particles)
	{
		/**
		 * GRAVITY APPLICATION:
		 * Reduces Z velocity component
		 * Negative = downward
		 * 9.81 m/s² = 981 cm/s² = standard Earth gravity
		 * 
		 * This is "per-frame" application
		 * Not Euler integration (no position update from gravity here)
		 * Velocity updated, then position updated below
		 */

		Particle.Velocity.Z -= 9.81f * DeltaTime;

		/**
		 * FORCE APPLICATION (when ComputeForces fully implemented):
		 * Particle.Velocity += (Forces / Particle.Mass) * DeltaTime;
		 * 
		 * Currently placeholder (ComputeForces not complete)
		 * TODO: Uncomment when pressure/viscosity forces ready
		 */

		// TODO: Particle.Velocity += (ComputedForces / Particle.Mass) * DeltaTime;

		/**
		 * POSITION UPDATE:
		 * Euler integration (velocity × time step)
		 * Most straightforward approach
		 * Sufficient for splash effects
		 * 
		 * Higher order methods (RK2, RK4) would be more accurate
		 * But added complexity not worth it for splashes
		 */

		Particle.Position += Particle.Velocity * DeltaTime;
	}

	/**
	 * PARTICLE CLEANUP:
	 * Remove particles that have fallen too far
	 * 
	 * Prevents:
	 * - Array growing unbounded
	 * - Infinite array searches
	 * - Memory leak over time
	 * 
	 * Represents:
	 * - Water drainage into ground
	 * - Evaporation
	 * - Physical boundary of simulation
	 * 
	 * Z threshold = -1000 UU
	 * With 100 UU/m = 10 meters below waterplane
	 * Particles have long time to fall and be visible
	 * But eventually cleaned up
	 * 
	 * OPTIMIZATION:
	 * RemoveAll with predicate is efficient
	 * Only removes particles matching condition
	 * Typical: 0-10 particles per frame (cleanup)
	 * 
	 * If splashes too fast disappeared:
	 * Increase threshold (e.g., -2000)
	 * If too many particles accumulate:
	 * Decrease threshold (e.g., -500)
	 */

	Particles.RemoveAll([](const FSPHParticle& P) { return P.Position.Z < -1000.0f; });
}

	 * Needed to detect when actors enter water
	 * 
	 * Physics: Dynamic collision type
	 * Responds to physics queries
	 * Generates overlap events when other objects collide
	 */

	PrimaryComponentTick.TickInterval = 1.0f / 60.0f;
	PrimaryComponentTick.bCanEverTick = true;

	SetCollisionEnabled(ECC_WorldDynamic);
	SetCollisionObjectType(ECC_WorldDynamic);
	SetGenerateOverlapEvents(true);
}

void UHybridWaterPlaneComponent::BeginPlay()
{
	/**
	 * Game initialization - Create all three layers
	 * 
	 * TIMING:
	 * BeginPlay called when:
	 * - Level fully loaded
	 * - All actors spawned
	 * - Ready to start gameplay
	 * 
	 * Why here and not constructor?
	 * - Constructor may run before rendering system ready
	 * - BeginPlay runs after all systems initialized
	 * - GPU resources created safely
	 * 
	 * LAYER INITIALIZATION:
	 * 1. FFT Waves: Always created (always active)
	 * 2. SWE: Always created (LOD controls usage)
	 * 3. SPH: Created only if bEnableSPHParticles (designer toggle)
	 * 
	 * BeginInitResource():
	 * Queues GPU resource initialization to render thread
	 * Doesn't block game thread
	 * Resources ready by first frame (usually)
	 */

	Super::BeginPlay();

	// Layer 1: FFT Ocean Waves
	if (!FFTOceanWaves.IsValid())
	{
		FFTOceanWaves = MakeShared<FFFTOceanWaves>(FFTResolution);
		BeginInitResource(FFTOceanWaves.Get());
		UE_LOG(LogTemp, Warning, TEXT("FFT Ocean initialized (%dx%d)"), FFTResolution, FFTResolution);
	}

	// Layer 2: SWE Simulation
	if (!SWESimulation.IsValid())
	{
		SWESimulation = MakeShared<FShallowWaterSimulation>(SWEGridWidth, SWEGridHeight, SWECellSize);
		BeginInitResource(SWESimulation.Get());
		UE_LOG(LogTemp, Warning, TEXT("SWE Simulation initialized (%dx%d grid, %.0f cm cells)"), 
			SWEGridWidth, SWEGridHeight, SWECellSize);
	}

	// Layer 3: SPH Particles (optional)
	if (!SPHInteraction.IsValid() && bEnableSPHParticles)
	{
		SPHInteraction = MakeShared<FLocalSPHInteraction>(MaxSPHParticles);
		SPHInteraction->Initialize();
		UE_LOG(LogTemp, Warning, TEXT("SPH Particles initialized (max %d particles)"), MaxSPHParticles);
	}

	// Setup event delegates for actor tracking
	OnComponentBeginOverlap.AddDynamic(this, &UHybridWaterPlaneComponent::OnBeginActorOverlap);
	OnComponentEndOverlap.AddDynamic(this, &UHybridWaterPlaneComponent::OnEndActorOverlap);

	// Compute buoyancy update interval
	BuoyancyUpdateInterval = 1.0f / BuoyancyUpdateFrequency;

	UE_LOG(LogTemp, Warning, TEXT("Hybrid Water Plane initialized successfully"));
}

void UHybridWaterPlaneComponent::TickComponent(float DeltaTime, ELevelTick TickType, FActorComponentTickFunction* ThisTickFunction)
{
	/**
	 * Main game loop - Called 60 times per second at 60 FPS
	 * 
	 * SEQUENCE:
	 * 1. Call parent tick (housekeeping)
	 * 2. Update all layers (FFT, SWE, SPH)
	 * 3. Apply buoyancy forces (on schedule)
	 * 4. Debug visualization (if enabled)
	 * 
	 * PERFORMANCE BUDGET:
	 * Target: 7-8 ms (60 FPS frame time = 16.67 ms)
	 * Remaining: ~9 ms for rest of game
	 * 
	 * @param DeltaTime Frame time (e.g., 0.01667 at 60 FPS)
	 * @param TickType Type of tick (normal, physics, etc.)
	 * @param ThisTickFunction Tick function handle
	 */

	Super::TickComponent(DeltaTime, TickType, ThisTickFunction);

	// Update all three simulation layers
	UpdateLayers(DeltaTime);

	// Apply forces to actors in water (on schedule)
	TimeSinceLastBuoyancyUpdate += DeltaTime;
	if (TimeSinceLastBuoyancyUpdate >= BuoyancyUpdateInterval)
	{
		UpdateBuoyancyForces(TimeSinceLastBuoyancyUpdate);
		TimeSinceLastBuoyancyUpdate = 0.0f;
	}

	// Debug visualization (optional)
	if (bDebugDrawWaterHeight || bDebugDrawBuoyancy || bDebugDrawPerformance)
	{
		DrawDebugVisualization();
	}
}

void UHybridWaterPlaneComponent::EndPlay(const EEndPlayReason::Type EndPlayReason)
{
	/**
	 * Level cleanup - Tear down resources
	 * 
	 * Called when:
	 * - Level unloads
	 * - Game exits
	 * - Component destroyed
	 * 
	 * Cleanup steps:
	 * 1. Remove event delegates (prevent dangling pointers)
	 * 2. Release SPH particles (CPU memory)
	 * 3. FRenderResource automatically releases GPU textures
	 * 
	 * GPU cleanup automatic:
	 * - FRenderResource calls ReleaseRHI() on render thread
	 * - Textures freed by GPU driver
	 * - Reference counting prevents double-delete
	 */

	OnComponentBeginOverlap.RemoveDynamic(this, &UHybridWaterPlaneComponent::OnBeginActorOverlap);
	OnComponentEndOverlap.RemoveDynamic(this, &UHybridWaterPlaneComponent::OnEndActorOverlap);

	if (SPHInteraction.IsValid())
	{
		SPHInteraction.Reset();
	}

	Super::EndPlay(EndPlayReason);
}

void UHybridWaterPlaneComponent::UpdateLayers(float DeltaTime)
{
	/**
	 * Master orchestrator for all layers
	 * 
	 * EXECUTION ORDER:
	 * 1. FFT (always, ~1ms) - visual base
	 * 2. SWE (conditional, ~3-5ms) - physics response
	 * 3. SPH (conditional, ~1-2ms) - particle details
	 * 
	 * Each layer independent
	 * Can profile each separately
	 * Can disable individually for debugging
	 * 
	 * PROFILING:
	 * Each layer times itself
	 * Results logged if debug flag enabled
	 * Helps identify bottlenecks
	 * 
	 * DESIGNER ADJUSTMENTS:
	 * If performance poor:
	 * - Disable SWE (save 3-5ms)
	 * - Disable SPH (save 1-2ms)
	 * - Reduce FFT resolution (save 0.3-0.5ms)
	 * - Increase SWE distance (LOD helps)
	 * 
	 * @param DeltaTime Frame time in seconds
	 */

	double StartTime = FPlatformTime::Seconds();

	// Layer 1: FFT Waves (always update)
	{
		double LayerStart = FPlatformTime::Seconds();
		UpdateFFTLayer(DeltaTime);
		PerfMetrics.FFTUpdateTime = (FPlatformTime::Seconds() - LayerStart) * 1000.0;
	}

	// Layer 2: SWE Physics (with distance LOD)
	{
		double LayerStart = FPlatformTime::Seconds();
		UpdateSWELayer(DeltaTime);
		PerfMetrics.SWEUpdateTime = (FPlatformTime::Seconds() - LayerStart) * 1000.0;
	}

	// Layer 3: SPH Particles (if enabled)
	if (bEnableSPHParticles && SPHInteraction.IsValid())
	{
		double LayerStart = FPlatformTime::Seconds();
		UpdateSPHLayer(DeltaTime);
		PerfMetrics.SPHUpdateTime = (FPlatformTime::Seconds() - LayerStart) * 1000.0;
	}

	PerfMetrics.TotalFrameTime = (FPlatformTime::Seconds() - StartTime) * 1000.0;

	/**
	 * PERFORMANCE ALERT:
	 * If total time exceeds 8ms consistently:
	 * - We're eating into rest of game's budget
	 * - Need to optimize or reduce detail
	 * - Log warning to help identify issue
	 */

	if (bDebugDrawPerformance && PerfMetrics.TotalFrameTime > 8.0)
	{
		UE_LOG(LogTemp, Warning, TEXT("Water frame time %.1f ms (exceeds 8ms budget!)"), PerfMetrics.TotalFrameTime);
	}
}

void UHybridWaterPlaneComponent::UpdateFFTLayer(float DeltaTime)
{
	/**
	 * Update Layer 1: FFT Ocean Waves
	 * 
	 * ALWAYS RUNS:
	 * - Provides visual foundation
	 * - GPU-based (cheap)
	 * - Results always available for rendering
	 * 
	 * PARAMETERS FROM DESIGNER:
	 * - FFTAmplitude: Wave height
	 * - WindDirection: Wave direction
	 * - WindSpeed: Wave energy
	 * - ChoppinessScale: Visual appearance
	 * 
	 * All adjustable in real-time in editor
	 * Changes visible immediately
	 * 
	 * @param DeltaTime Frame time
	 */

	if (!FFTOceanWaves.IsValid()) return;

	// Package parameters
	FFFTWaveParameters Params;
	Params.Amplitude = FFTAmplitude;
	Params.WindDirection = WindDirection;
	Params.WindSpeed = WindSpeed;
	Params.ChoppinessScale = ChoppinessScale;
	Params.TimeScale = 1.0f;

	// Update GPU waves
	FFTOceanWaves->UpdateWaves(DeltaTime, Params);
}

void UHybridWaterPlaneComponent::UpdateSWELayer(float DeltaTime)
{
	/**
	 * Update Layer 2: Shallow Water Equations
	 * 
	 * LOD-CONTROLLED:
	 * - Only simulate if camera within SWESimulationDistance
	 * - Saves 3-5ms per frame when not visible
	 * - Player won't notice ripples beyond ~50m
	 * 
	 * DISTANCE CHECK:
	 * Get player camera location
	 * Compare to water plane center
	 * Skip SWE if too far
	 * 
	 * @param DeltaTime Frame time
	 */

	if (!SWESimulation.IsValid()) return;

	// Distance-based LOD
	APlayerController* PC = GetWorld()->GetFirstPlayerController();
	if (PC)
	{
		FVector CameraLocation = PC->PlayerCameraManager->GetCameraLocation();
		float DistanceToCamera = FVector::Dist(GetComponentLocation(), CameraLocation);

		if (DistanceToCamera <= SWESimulationDistance)
		{
			// Camera close - simulate full SWE physics
			SWESimulation->UpdateSimulation(DeltaTime);
		}
		// else: Camera far - use only FFT (already updated)
	}
}

void UHybridWaterPlaneComponent::UpdateSPHLayer(float DeltaTime)
{
	/**
	 * Update Layer 3: SPH Particles
	 * 
	 * CONDITIONS TO UPDATE:
	 * 1. bEnableSPHParticles = true (designer toggle)
	 * 2. SPHInteraction created (happens if enabled)
	 * 3. Could add distance check (particles not visible beyond ~30m)
	 * 
	 * @param DeltaTime Frame time
	 */

	if (!SPHInteraction.IsValid()) return;

	// Update particle simulation
	SPHInteraction->UpdateParticles(DeltaTime, GetComponentLocation());
}

void UHybridWaterPlaneComponent::UpdateBuoyancyForces(float DeltaTime)
{
	/**
	 * Apply buoyancy and drag forces to actors in water
	 * 
	 * SCHEDULING:
	 * Called every BuoyancyUpdateFrequency (default 60 Hz)
	 * Not necessarily every frame (could be 30 Hz for performance)
	 * 
	 * FORCE TYPES:
	 * 1. Buoyancy: Upward force proportional to displaced volume
	 * 2. Drag: Resistance to movement through water
	 * 3. SPH forces: Splash particle interactions
	 * 
	 * For each actor in water:
	 * - Calculate submerged volume
	 * - Calculate buoyancy force
	 * - Calculate drag force
	 * - Get SPH splash forces
	 * - Apply combined force to physics
	 * 
	 * @param DeltaTime Actual time since last update
	 */

	double StartTime = FPlatformTime::Seconds();

	// Apply forces to all actors currently in water
	for (AActor* Actor : ActorsInWater)
	{
		if (Actor && !Actor->IsHidden())
		{
			ApplyBuoyancyToActor(Actor, DeltaTime);
		}
	}

	PerfMetrics.BuoyancyUpdateTime = (FPlatformTime::Seconds() - StartTime) * 1000.0;
}

void UHybridWaterPlaneComponent::ApplyBuoyancyToActor(AActor* Actor, float DeltaTime)
{
	/**
	 * Apply physics forces to a single actor in water
	 * 
	 * FORCES APPLIED:
	 * 1. BUOYANCY (Archimedes' Principle)
	 *    F = ρ * g * V_submerged
	 *    Upward force proportional to water displaced
	 *    Makes objects float
	 * 
	 * 2. DRAG (Viscosity)
	 *    F = -c * (v_actor - v_water) * V_submerged
	 *    Opposes motion through water
	 *    Slows actors down
	 *    Pushes actors along current
	 * 
	 * 3. SPH FORCES (Splash interactions)
	 *    From nearby splash particles
	 *    Pushes actors around by splashes
	 * 
	 * COMBINED EFFECT:
	 * Actors experience realistic water physics
	 * Float when buoyant
	 * Sink when too heavy
	 * Get pushed by splashes
	 * Slow down from drag
	 * 
	 * @param Actor The actor to affect
	 * @param DeltaTime Time since last update
	 * 
	 * DESIGNER TUNING:
	 * - WaterDensity: Higher = stronger buoyancy (easier to float)
	 * - ActorSubmergedThreshold: Skip forces below this volume
	 * - Drag coefficient: (0.5 hardcoded, could be parameter)
	 */

	// Get physics component (must have physics enabled)
	UPrimitiveComponent* PrimComp = Cast<UPrimitiveComponent>(Actor->GetRootComponent());
	if (!PrimComp || !PrimComp->IsSimulatingPhysics())
	{
		return;  // Non-physics actor, skip
	}

	float ActorMass = PrimComp->GetMass();
	float SubmergedVolume = CalculateSubmergedVolume(Actor);

	// Skip if barely submerged (optimization)
	if (SubmergedVolume < ActorSubmergedThreshold)
	{
		return;
	}

	// === BUOYANCY FORCE ===
	float BuoyancyMagnitude = WaterDensity * 9.81f * SubmergedVolume;
	FVector BuoyancyForce = FVector::UpVector * BuoyancyMagnitude;

	// === DRAG FORCE ===
	FVector ActorVelocity = PrimComp->GetPhysicsLinearVelocity();
	FVector WaterVelocity = GetWaterVelocityAt(Actor->GetActorLocation());
	FVector VelocityDifference = ActorVelocity - WaterVelocity;
	FVector DragForce = -VelocityDifference * 0.5f * SubmergedVolume;

	// === SPH FORCES ===
	FVector SPHForce = FVector::ZeroVector;
	if (SPHInteraction.IsValid() && SPHInteraction->HasActiveParticles())
	{
		SPHForce = SPHInteraction->GetInteractionForceAt(Actor->GetActorLocation(), 500.0f);
	}

	// === APPLY COMBINED FORCES ===
	FVector TotalForce = BuoyancyForce + DragForce + SPHForce;
	PrimComp->AddForce(TotalForce);

	// Store for debugging/visualization
	ActorWaterForces[Actor] = TotalForce;
	ActorSubmergedVolumes[Actor] = SubmergedVolume;
}

float UHybridWaterPlaneComponent::CalculateSubmergedVolume(AActor* Actor) const
{
	/**
	 * Calculate how much of an actor is underwater
	 * 
	 * ALGORITHM: Bounding box intersection with water plane
	 * 
	 * APPROXIMATION NOTE:
	 * Uses bounding box (simple, fast)
	 * Could use convex hull (more accurate, slower)
	 * For game purposes: Bounding box sufficient
	 * Players won't notice difference
	 * 
	 * CASES:
	 * 1. Fully submerged: Volume = total box volume
	 * 2. Not submerged: Volume = 0
	 * 3. Partially submerged: Volume = proportional
	 * 
	 * @param Actor The actor to measure
	 * @return Submerged volume in cubic cm
	 */

	UPrimitiveComponent* PrimComp = Cast<UPrimitiveComponent>(Actor->GetRootComponent());
	if (!PrimComp) return 0.0f;

	// Get actor bounds
	FVector ActorCenter = Actor->GetActorLocation();
	FVector Extent = PrimComp->Bounds.BoxExtent;  // Half-extents
	float WaterHeight = GetComponentLocation().Z;

	// Calculate Z range of bounding box
	float ActorLowestZ = ActorCenter.Z - Extent.Z;
	float ActorHighestZ = ActorCenter.Z + Extent.Z;

	// Determine submersion case
	if (ActorHighestZ < WaterHeight)
	{
		// Fully submerged
		return Extent.X * Extent.Y * Extent.Z * 8.0f;  // V = (2X)*(2Y)*(2Z)
	}
	else if (ActorLowestZ > WaterHeight)
	{
		// Not submerged
		return 0.0f;
	}
	else
	{
		// Partially submerged
		float SubmergedHeight = WaterHeight - ActorLowestZ;
		float MaxHeight = ActorHighestZ - ActorLowestZ;
		float SubmergedRatio = FMath::Clamp(SubmergedHeight / MaxHeight, 0.0f, 1.0f);

		return Extent.X * Extent.Y * Extent.Z * 8.0f * SubmergedRatio;
	}
}

void UHybridWaterPlaneComponent::OnBeginActorOverlap(UPrimitiveComponent* OverlappedComponent, AActor* OtherActor,
	UPrimitiveComponent* OtherComp, int32 OtherBodyIndex, bool bFromSweep, const FHitResult& SweepResult)
{
	/**
	 * Callback when actor enters water volume
	 * 
	 * CALLED BY:
	 * UE's collision system automatically
	 * When actor component overlaps water collision volume
	 * 
	 * PURPOSE:
	 * Track which actors need buoyancy forces
	 * Allocate storage for volume/force tracking
	 * 
	 * @param OverlappedComponent This component (water)
	 * @param OtherActor Actor entering (e.g., player, boat)
	 * @param OtherComp Component of actor entering
	 * @param OtherBodyIndex Physics body index (usually 0)
	 * @param bFromSweep Whether overlap was from sweep (movement) or not
	 * @param SweepResult Details of overlap (if from sweep)
	 * 
	 * DESIGNER HOOKS:
	 * Could add effects here:
	 * - Play splash sound
	 * - Spawn splash particles
	 * - Apply damage (toxic water, lava)
	 * - Trigger gameplay logic
	 * 
	 * Currently just adds to tracking sets
	 */

	if (OtherActor)
	{
		ActorsInWater.Add(OtherActor);
		ActorSubmergedVolumes.Add(OtherActor, 0.0f);
	}
}

void UHybridWaterPlaneComponent::OnEndActorOverlap(UPrimitiveComponent* OverlappedComponent, AActor* OtherActor,
	UPrimitiveComponent* OtherComp, int32 OtherBodyIndex)
{
	/**
	 * Callback when actor leaves water volume
	 * 
	 * CALLED BY:
	 * UE's collision system automatically
	 * When actor component exits water collision volume
	 * 
	 * PURPOSE:
	 * Stop applying forces to actor
	 * Clean up tracking structures
	 * Prevent stale data
	 * 
	 * @param OverlappedComponent This component (water)
	 * @param OtherActor Actor exiting
	 * @param OtherComp Component of actor exiting
	 * @param OtherBodyIndex Physics body index
	 * 
	 * DESIGNER HOOKS:
	 * Could add effects here:
	 * - Play water exit sound
	 * - Spawn dripping particles
	 * - Trigger exit logic (stopped swimming)
	 * 
	 * Currently just removes from tracking
	 */

	if (OtherActor)
	{
		ActorsInWater.Remove(OtherActor);
		ActorSubmergedVolumes.Remove(OtherActor);
		ActorWaterForces.Remove(OtherActor);
	}
}

/**
 * Performance-optimized structure for spatial hashing
 * Pattern 1: Spatial Hashing for Fast Actor Queries
 */
struct FWaterSpatialHashGrid
{
	static constexpr int32 GridCellSize = 500; // Unreal units per cell
	static constexpr int32 MaxActorsPerCell = 32;

	struct FCellData
	{
		TArray<AActor*> Actors;
		FVector2D CellCenter;
	};

	TMap<int64, FCellData> Grid;

	int64 GetCellKey(const FVector& WorldPosition) const
	{
		int32 X = FMath::FloorToInt(WorldPosition.X / GridCellSize);
		int32 Y = FMath::FloorToInt(WorldPosition.Y / GridCellSize);
		return ((int64)X << 32) | (uint32)Y;
	}

	void AddActorToGrid(AActor* Actor)
	{
		if (!Actor) return;
		int64 Key = GetCellKey(Actor->GetActorLocation());
		FCellData& Cell = Grid.FindOrAdd(Key);
		if (Cell.Actors.Num() < MaxActorsPerCell)
		{
			Cell.Actors.AddUnique(Actor);
		}
	}

	void RemoveActorFromGrid(AActor* Actor)
	{
		if (!Actor) return;
		int64 Key = GetCellKey(Actor->GetActorLocation());
		if (FCellData* Cell = Grid.Find(Key))
		{
			Cell->Actors.Remove(Actor);
		}
	}

	void GetActorsNearLocation(const FVector& Location, float Radius, TArray<AActor*>& OutActors)
	{
		int32 SearchRadius = FMath::CeilToInt(Radius / GridCellSize) + 1;
		int32 CenterX = FMath::FloorToInt(Location.X / GridCellSize);
		int32 CenterY = FMath::FloorToInt(Location.Y / GridCellSize);

		for (int32 X = CenterX - SearchRadius; X <= CenterX + SearchRadius; ++X)
		{
			for (int32 Y = CenterY - SearchRadius; Y <= CenterY + SearchRadius; ++Y)
			{
				int64 Key = ((int64)X << 32) | (uint32)Y;
				if (FCellData* Cell = Grid.Find(Key))
				{
					OutActors.Append(Cell->Actors);
				}
			}
		}
	}
};

// Wave Simulation Compute Shader
class FWaveSimulationCS : public FGlobalShader
{
public:
	DECLARE_GLOBAL_SHADER(FWaveSimulationCS);
	SHADER_USE_PARAMETER_STRUCT(FWaveSimulationCS, FGlobalShader);

	BEGIN_SHADER_PARAMETER_STRUCT(FParameters, )
		SHADER_PARAMETER_RDG_TEXTURE_UAV(RWTexture2D<float4>, OutWaveHeight)
		SHADER_PARAMETER_RDG_TEXTURE(Texture2D, InWaveHeight)
		SHADER_PARAMETER(FVector2f, ImpactLocation)
		SHADER_PARAMETER(float, ImpactForce)
		SHADER_PARAMETER(float, DeltaTime)
		SHADER_PARAMETER(uint32, TextureResolution)
	END_SHADER_PARAMETER_STRUCT()

	static bool ShouldCompilePermutation(const FGlobalShaderPermutationParameters& Parameters)
	{
		return IsFeatureLevelSupported(Parameters.Platform, ERHIFeatureLevel::SM5);
	}

	static void ModifyCompilationEnvironment(const FGlobalShaderPermutationParameters& Parameters, 
		FShaderCompilerEnvironment& OutEnvironment)
	{
		FGlobalShader::ModifyCompilationEnvironment(Parameters, OutEnvironment);
		OutEnvironment.SetDefine(TEXT("THREADGROUP_SIZEX"), 16);
		OutEnvironment.SetDefine(TEXT("THREADGROUP_SIZEY"), 16);
	}
};

IMPLEMENT_GLOBAL_SHADER(FWaveSimulationCS, TEXT("/Shaders/Private/WaveSimulation.usf"), TEXT("WaveSimulationCS"), SF_Compute);

FWaveSimulationResource::FWaveSimulationResource(int32 InResolution)
	: Resolution(InResolution)
{
}

FWaveSimulationResource::~FWaveSimulationResource()
{
}

void FWaveSimulationResource::InitializeRHI(FRHICommandListBase& RHICmdList)
{
	// Create wave height texture
	FRHITextureCreateDesc Desc =
		FRHITextureCreateDesc::Create2D(TEXT("WaveHeightTexture"))
		.SetExtent(Resolution, Resolution)
		.SetFormat(PF_FloatRGBA)
		.SetFlags(ETextureCreateFlags::RenderTargetable | ETextureCreateFlags::ShaderResource | ETextureCreateFlags::UAV);

	WaveHeightTexture = RHICreateTexture(Desc);
	WaveHeightTextureTemp = RHICreateTexture(Desc);

	// Create wave normal texture
	Desc.SetDimensions(Resolution, Resolution, 1);
	WaveNormalTexture = RHICreateTexture(Desc);
}

void FWaveSimulationResource::ReleaseRHI()
{
	WaveHeightTexture.SafeRelease();
	WaveHeightTextureTemp.SafeRelease();
	WaveNormalTexture.SafeRelease();
}

void FWaveSimulationResource::AddWaveImpact(const FVector& ImpactLocation, float ImpactForce)
{
	FScopeLock Lock(&ImpactLock);
	PendingImpacts.Add(ImpactLocation);
}

void FWaveSimulationResource::UpdateWaves(FRDGBuilder& GraphBuilder, float DeltaTime, const TArray<FVector>& ImpactLocations)
{
	if (!WaveHeightTexture.IsValid()) return;

	FRDGTextureRef WaveHeightRDG = GraphBuilder.RegisterExternalTexture(
		CreateRenderTarget(WaveHeightTexture, TEXT("WaveHeight")));

	{
		FScopeLock Lock(&ImpactLock);
		PendingImpacts.Reset();
	}

	// Process each impact with a compute shader pass
	for (const FVector& ImpactLocation : ImpactLocations)
	{
		auto* PassParameters = GraphBuilder.AllocParameters<FWaveSimulationCS::FParameters>();
		PassParameters->OutWaveHeight = GraphBuilder.CreateUAV(WaveHeightRDG);
		PassParameters->InWaveHeight = WaveHeightRDG;
		PassParameters->ImpactLocation = FVector2f(ImpactLocation.X, ImpactLocation.Y);
		PassParameters->ImpactForce = 1.0f; // Normalized impact
		PassParameters->DeltaTime = DeltaTime;
		PassParameters->TextureResolution = Resolution;

		FComputeShaderUtils::AddPass(
			GraphBuilder,
			RDG_EVENT_NAME("WaveSimulation"),
			ComputeShader,
			PassParameters,
			FIntPoint(Resolution / 16, Resolution / 16));
	}
}

UVolumetricWaterPlaneComponent::UVolumetricWaterPlaneComponent()
{
	PrimaryComponentTick.TickInterval = 1.0f / 60.0f;
	PrimaryComponentTick.bCanEverTick = true;

	SetCollisionEnabled(ECC_WorldDynamic);
	SetCollisionObjectType(ECC_WorldDynamic);
	SetGenerateOverlapEvents(true);
}

void UVolumetricWaterPlaneComponent::BeginPlay()
{
	Super::BeginPlay();

	// Find water subsystem
	if (UWorld* World = GetWorld())
	{
		if (UWaterSubsystem* WaterSubsystem = UWaterSubsystem::GetWaterSubsystem(World))
		{
			BuoyancyManager = WaterSubsystem->GetBuoyancyManager();
		}
	}

	// Initialize wave simulation
	InitializeWaveSimulation();

	// Setup overlap events
	OnComponentBeginOverlap.AddDynamic(this, &UVolumetricWaterPlaneComponent::OnBeginWaterOverlap);
	OnComponentEndOverlap.AddDynamic(this, &UVolumetricWaterPlaneComponent::OnEndWaterOverlap);

	// Set initial water plane properties
	WaterPlanePosition = GetComponentLocation();
	WaterPlaneNormal = GetComponentRotation().Vector();
	WaterBaseHeight = GetComponentLocation().Z;
}

void UVolumetricWaterPlaneComponent::OnRegister()
{
	Super::OnRegister();

	// Create wave simulation resource
	if (!WaveSimulationResource.IsValid())
	{
		WaveSimulationResource = MakeShared<FWaveSimulationResource>(WaveSimulationResolution);
		BeginInitResource(WaveSimulationResource.Get());
	}
}

void UVolumetricWaterPlaneComponent::OnUnregister()
{
	if (WaveSimulationResource.IsValid())
	{
		BeginReleaseResource(WaveSimulationResource.Get());
		FlushRenderingCommands();
		WaveSimulationResource.Reset();
	}

	Super::OnUnregister();
}

void UVolumetricWaterPlaneComponent::TickComponent(float DeltaTime, ELevelTick TickType, FActorComponentTickFunction* ThisTickFunction)
{
	Super::TickComponent(DeltaTime, TickType, ThisTickFunction);

	// Pattern 1: Update spatial hash periodically
	UpdateSpatialHash();

	// Pattern 2: Update wave simulation on schedule
	UpdateWaveSimulation(DeltaTime);

	// Pattern 3: Update buoyancy on schedule
	UpdateBuoyancy(DeltaTime);

	// Debug visualization
	if (bDebugDrawSpatialHash || bDebugDrawBuoyancy)
	{
		DrawDebugVisualization();
	}
}

void UVolumetricWaterPlaneComponent::EndPlay(const EEndPlayReason::Type EndPlayReason)
{
	OnComponentBeginOverlap.RemoveDynamic(this, &UVolumetricWaterPlaneComponent::OnBeginWaterOverlap);
	OnComponentEndOverlap.RemoveDynamic(this, &UVolumetricWaterPlaneComponent::OnEndWaterOverlap);

	Super::EndPlay(EndPlayReason);
}

// ============================================================================
// Pattern 1: Spatial Hashing
// ============================================================================

void UVolumetricWaterPlaneComponent::UpdateSpatialHash()
{
	float CurrentTime = GetWorld()->GetTimeSeconds();
	float UpdateInterval = 1.0f / 30.0f; // Update 30 times per second

	if (CurrentTime - LastSpatialHashUpdate < UpdateInterval)
	{
		return;
	}

	LastSpatialHashUpdate = CurrentTime;

	// Rebuild relevant cells
	for (AActor* Actor : ActorsInWater)
	{
		if (Actor && !Actor->IsHidden())
		{
			SpatialHashGrid.RemoveActorFromGrid(Actor);
			SpatialHashGrid.AddActorToGrid(Actor);
		}
	}
}

void UVolumetricWaterPlaneComponent::RegisterActorWithWater(AActor* Actor)
{
	if (!Actor) return;

	ActorsInWater.Add(Actor);
	SpatialHashGrid.AddActorToGrid(Actor);
	ActorSubmergedVolumes.Add(Actor, 0.0f);
}

void UVolumetricWaterPlaneComponent::UnregisterActorFromWater(AActor* Actor)
{
	if (!Actor) return;

	ActorsInWater.Remove(Actor);
	SpatialHashGrid.RemoveActorFromGrid(Actor);
	ActorSubmergedVolumes.Remove(Actor);
}

void UVolumetricWaterPlaneComponent::OnBeginWaterOverlap(UPrimitiveComponent* OverlappedComponent, AActor* OtherActor,
	UPrimitiveComponent* OtherComp, int32 OtherBodyIndex, bool bFromSweep, const FHitResult& SweepResult)
{
	RegisterActorWithWater(OtherActor);
}

void UVolumetricWaterPlaneComponent::OnEndWaterOverlap(UPrimitiveComponent* OverlappedComponent, AActor* OtherActor,
	UPrimitiveComponent* OtherComp, int32 OtherBodyIndex)
{
	UnregisterActorFromWater(OtherActor);
}

void UVolumetricWaterPlaneComponent::UpdateWaveSimulation(float DeltaTime)
{
	TimeSinceLastWaveUpdate += DeltaTime;
	float UpdateInterval = 1.0f / WaveSimulationFrequency;

	if (TimeSinceLastWaveUpdate < UpdateInterval)
	{
		return;
	}

	WaveSimulationDeltaTime = TimeSinceLastWaveUpdate;
	TimeSinceLastWaveUpdate = 0.0f;

	// Only simulate waves within range of camera for LOD
	APlayerController* PC = GetWorld()->GetFirstPlayerController();
	if (!PC) return;

	FVector CameraLocation = PC->PlayerCameraManager->GetCameraLocation();
	float DistanceToCamera = FVector::Dist(GetComponentLocation(), CameraLocation);

	// LOD: Skip simulation if too far away
	if (DistanceToCamera > WaveSimulationDistance)
	{
		return;
	}

	// Collect impact locations
	TArray<FVector> ImpactLocations;
	for (AActor* Actor : ActorsInWater)
	{
		if (Actor && Actor->IsActorTickEnabled())
		{
			// Only add impacts from fast-moving objects
			if (UPrimitiveComponent* PrimComp = Cast<UPrimitiveComponent>(Actor->GetRootComponent()))
			{
				FVector Velocity = PrimComp->GetPhysicsLinearVelocity();
				if (Velocity.Length() > 100.0f) // Minimum velocity threshold
				{
					ImpactLocations.Add(Actor->GetActorLocation());
				}
			}
		}
	}

	// Schedule GPU wave simulation
	SimulateWavesGPU(nullptr); // Pass FRDGBuilder when called from render thread
}

// Pattern 3: Physics and Buoyancy
// ============================================================================

void UVolumetricWaterPlaneComponent::UpdateBuoyancy(float DeltaTime)
{
	TimeSinceLastBuoyancyUpdate += DeltaTime;
	float UpdateInterval = 1.0f / BuoyancyUpdateFrequency;

	if (TimeSinceLastBuoyancyUpdate < UpdateInterval)
	{
		return;
	}

	TimeSinceLastBuoyancyUpdate = 0.0f;

	// Update buoyancy for all actors in water
	for (AActor* Actor : ActorsInWater)
	{
		if (Actor && !Actor->IsHidden())
		{
			ApplyBuoyancyForceToActor(Actor, DeltaTime);
		}
	}
}

void UVolumetricWaterPlaneComponent::ApplyBuoyancyForceToActor(AActor* Actor, float DeltaTime)
{
	if (!Actor) return;

	UPrimitiveComponent* PrimComp = Cast<UPrimitiveComponent>(Actor->GetRootComponent());
	if (!PrimComp || !PrimComp->IsSimulatingPhysics())
	{
		return;
	}

	// Get actor properties
	float ActorMass = PrimComp->GetMass();
	if (ActorMass < MinimumMassForBuoyancy)
	{
		return;
	}

	FVector ActorLocation = Actor->GetActorLocation();
	float SubmergedVolume = CalculateSubmergedVolume(Actor, WaterPlanePosition);

	// Buoyancy force = water density * gravity * submerged volume
	float BuoyancyForce = WaterDensity * 981.0f * SubmergedVolume; // 981 cm/s^2 gravity

	// Apply buoyancy (upward force)
	FVector BuoyancyVector = FVector::UpVector * BuoyancyForce;
	PrimComp->AddForce(BuoyancyVector);

	// Apply water resistance/viscosity
	FVector Velocity = PrimComp->GetPhysicsLinearVelocity();
	FVector ResistanceForce = -Velocity * WaterViscosity * SubmergedVolume;
	PrimComp->AddForce(ResistanceForce);

	// Store for debugging
	ActorSubmergedVolumes[Actor] = SubmergedVolume;
}

float UVolumetricWaterPlaneComponent::CalculateSubmergedVolume(AActor* Actor, const FVector& WaterSurfaceLocation)
{
	if (!Actor) return 0.0f;

	UPrimitiveComponent* PrimComp = Cast<UPrimitiveComponent>(Actor->GetRootComponent());
	if (!PrimComp) return 0.0f;

	// Get actor bounds
	FVector ActorCenter = Actor->GetActorLocation();
	FVector Extent = PrimComp->Bounds.BoxExtent;

	// Simple approximation: check how much of the bounding box is submerged
	float ActorLowestZ = ActorCenter.Z - Extent.Z;
	float ActorHighestZ = ActorCenter.Z + Extent.Z;

	if (ActorHighestZ < WaterPlanePosition.Z)
	{
		// Fully submerged
		return Extent.X * Extent.Y * Extent.Z * 8.0f; // Approximate volume
	}
	else if (ActorLowestZ > WaterPlanePosition.Z)
	{
		// Not submerged
		return 0.0f;
	}
	else
	{
		// Partially submerged
		float SubmergedHeight = WaterPlanePosition.Z - ActorLowestZ;
		float MaxHeight = ActorHighestZ - ActorLowestZ;
		float SubmergedRatio = FMath::Clamp(SubmergedHeight / MaxHeight, 0.0f, 1.0f);

		return Extent.X * Extent.Y * Extent.Z * 8.0f * SubmergedRatio;
	}
}

FVector UVolumetricWaterPlaneComponent::GetWaterVelocityAtLocation(const FVector& Location)
{
	// Simple implementation: return constant velocity
	// In a more advanced system, this would query wave simulation results
	return FVector::ZeroVector;
}

bool UVolumetricWaterPlaneComponent::GetWaterSurfaceInfoAtLocation(
	const FVector& InWorldLocation,
	FVector& OutWaterSurfaceLocation,
	FVector& OutWaterNormal,
	float& OutWaterDepth)
{
	OutWaterSurfaceLocation = WaterPlanePosition;
	OutWaterNormal = WaterPlaneNormal;
	OutWaterDepth = FMath::Max(0.0f, WaterPlanePosition.Z - InWorldLocation.Z);

	return OutWaterDepth >= 0.0f;
}

void UVolumetricWaterPlaneComponent::ApplyWaveImpact(const FVector& ImpactLocation, float ImpactForce, float ImpactRadius)
{
	if (WaveSimulationResource.IsValid())
	{
		WaveSimulationResource->AddWaveImpact(ImpactLocation, ImpactForce);
	}

	// Apply immediate force to nearby actors (for splash/push effect)
	TArray<AActor*> NearbyActors;
	SpatialHashGrid.GetActorsNearLocation(ImpactLocation, ImpactRadius, NearbyActors);

	for (AActor* Actor : NearbyActors)
	{
		if (Actor)
		{
			if (UPrimitiveComponent* PrimComp = Cast<UPrimitiveComponent>(Actor->GetRootComponent()))
			{
				FVector Direction = (Actor->GetActorLocation() - ImpactLocation).GetSafeNormal();
				float Distance = FVector::Dist(Actor->GetActorLocation(), ImpactLocation);
				float FalloffForce = ImpactForce * (1.0f - FMath::Clamp(Distance / ImpactRadius, 0.0f, 1.0f));

				PrimComp->AddImpulse(Direction * FalloffForce);
			}
		}
	}
}

void UVolumetricWaterPlaneComponent::DrawDebugVisualization()
{
	if (bDebugDrawSpatialHash)
	{
		// Draw spatial grid cells
		for (const auto& Cell : SpatialHashGrid.Grid)
		{
			int32 X = (int32)(Cell.Key >> 32);
			int32 Y = (int32)(Cell.Key & 0xFFFFFFFF);

			FVector CellMin = FVector(X * FWaterSpatialHashGrid::GridCellSize, Y * FWaterSpatialHashGrid::GridCellSize, GetComponentLocation().Z - 100);
			FVector CellMax = FVector((X + 1) * FWaterSpatialHashGrid::GridCellSize, (Y + 1) * FWaterSpatialHashGrid::GridCellSize, GetComponentLocation().Z + 100);

			DrawDebugBox(GetWorld(), (CellMin + CellMax) * 0.5f, (CellMax - CellMin) * 0.5f, FColor::Cyan, false, -1.0f, 0, 2.0f);

			// Draw actors in cell
			for (AActor* Actor : Cell.Value.Actors)
			{
				if (Actor)
				{
					DrawDebugPoint(GetWorld(), Actor->GetActorLocation(), 10.0f, FColor::Green, false, -1.0f, 0);
				}
			}
		}
	}

	if (bDebugDrawBuoyancy)
	{
		// Draw buoyancy forces
		for (const auto& Pair : ActorSubmergedVolumes)
		{
			AActor* Actor = Pair.Key;
			float SubmergedVolume = Pair.Value;

			if (Actor)
			{
				FVector ActorLoc = Actor->GetActorLocation();
				float BuoyancyForce = WaterDensity * 981.0f * SubmergedVolume;

				DrawDebugArrow(GetWorld(), ActorLoc, ActorLoc + FVector::UpVector * BuoyancyForce * 0.01f, 5.0f, FColor::Blue, false, -1.0f, 0);
				DrawDebugString(GetWorld(), ActorLoc + FVector::UpVector * 50, FString::Printf(TEXT("Vol: %.2f"), SubmergedVolume), nullptr, FColor::White, -1.0f, false);
			}
		}
	}
}
